WITH candidates AS (
  SELECT sp.child_agent_id,
         sp.requester_source,
         COALESCE(child.status, sp.status) AS child_status,
         parent.status AS parent_status,
         sp.completed_at,
         CASE
           WHEN sp.requester_source = 'scheduler.cron'
             THEN 'scheduler_cron_stale_root'
           ELSE 'old_terminal_non_cron_root'
         END AS prune_reason
  FROM subagent_spawns AS sp
  JOIN agents AS parent
    ON parent.id = sp.parent_agent_id
  LEFT JOIN agents AS child
    ON child.id = sp.child_agent_id
  WHERE sp.completed_at IS NOT NULL
    AND sp.seen_at IS NULL
    AND parent.kind = 'root'
    AND (
      (
        sp.requester_source = 'scheduler.cron'
        AND sp.created_at <= $1
        AND sp.completed_at <= $1
        AND (
          parent.updated_at < sp.completed_at
          OR parent.status IN ('completed', 'errored', 'shutdown', 'not_found')
        )
      )
      OR (
        sp.requester_source IN ('runtime', 'runtime.browser_monitor')
        AND sp.created_at <= $2
        AND sp.completed_at <= $2
        AND COALESCE(child.status, sp.status) IN ('shutdown', 'interrupted', 'errored')
      )
    )
    AND NOT EXISTS (
      SELECT 1
      FROM subagent_spawns AS descendant
      WHERE descendant.parent_agent_id = sp.child_agent_id
        AND descendant.completed_at IS NULL
    )
    AND NOT EXISTS (
      SELECT 1
      FROM subagent_spawns AS sibling
      WHERE sibling.parent_agent_id = sp.parent_agent_id
        AND sibling.completed_at IS NULL
    )
    AND NOT EXISTS (
      SELECT 1
      FROM subagent_progress_tool_events AS tool_event
      WHERE tool_event.child_agent_id = sp.child_agent_id
        AND tool_event.delivered_at IS NULL
    )
    AND NOT EXISTS (
      SELECT 1
      FROM subagent_progress_message_events AS message_event
      WHERE message_event.child_agent_id = sp.child_agent_id
        AND message_event.delivered_at IS NULL
    )
    AND NOT EXISTS (
      SELECT 1
      FROM subagent_monitor_decisions AS decision
      WHERE decision.child_agent_id = sp.child_agent_id
        AND decision.created_at >= $3
    )
  ORDER BY sp.created_at ASC, sp.child_agent_id ASC
  LIMIT $4
  FOR UPDATE OF sp SKIP LOCKED
),
updated AS (
  UPDATE subagent_spawns AS sp
  SET seen_at = $5
  FROM candidates AS c
  WHERE sp.child_agent_id = c.child_agent_id
    AND sp.completed_at IS NOT NULL
    AND sp.seen_at IS NULL
  RETURNING c.requester_source,
            c.child_status,
            c.parent_status,
            c.completed_at,
            c.prune_reason
)

WITH window_start AS (
  SELECT coalesce(max(event_seq), 0) AS event_seq FROM (
    (SELECT event_seq FROM runtime.events
     WHERE parent_agent_id = $1 AND event_name = 'message.user'
       AND NULLIF(TRIM(agent_id), '') IS NULL AND created_at < to_timestamp($2)
     ORDER BY event_seq DESC LIMIT 1)
    UNION ALL
    (SELECT event_seq FROM runtime.events
     WHERE agent_id = $1 AND event_name = 'message.user' AND created_at < to_timestamp($2)
     ORDER BY event_seq DESC LIMIT 1)
  ) AS before_window
),
request_messages AS (
  SELECT event_seq,
         (replace(payload_json, '\u0000', '\ufffd')::jsonb #>> '{metadata,context_seq}')::bigint AS context_seq
  FROM (
    SELECT e.event_seq, e.payload_json FROM runtime.events AS e, window_start
    WHERE e.parent_agent_id = $1 AND NULLIF(TRIM(e.agent_id), '') IS NULL
      AND e.event_name IN ('message.user', 'message.assistant', 'message.internal')
      AND e.event_seq >= window_start.event_seq AND e.request_id = $3
    UNION ALL
    SELECT e.event_seq, e.payload_json FROM runtime.events AS e, window_start
    WHERE e.agent_id = $1
      AND e.event_name IN ('message.user', 'message.assistant', 'message.internal')
      AND e.event_seq >= window_start.event_seq AND e.request_id = $3
  ) AS request_events
),
span AS (
  SELECT min(event_seq) AS first_event_seq,
         min(context_seq) AS first_context_seq, max(context_seq) AS last_context_seq
  FROM request_messages WHERE context_seq IS NOT NULL
),
reply_before AS (
  SELECT payload_json FROM (
    (SELECT e.event_seq, e.payload_json FROM runtime.events AS e, span
     WHERE e.parent_agent_id = $1 AND NULLIF(TRIM(e.agent_id), '') IS NULL
       AND e.event_seq < span.first_event_seq AND e.event_name = 'message.assistant'
       AND (replace(e.payload_json, '\u0000', '\ufffd')::jsonb #>> '{metadata,context_seq}') IS NOT NULL
     ORDER BY e.event_seq DESC LIMIT 1)
    UNION ALL
    (SELECT e.event_seq, e.payload_json FROM runtime.events AS e, span
     WHERE e.agent_id = $1 AND e.event_seq < span.first_event_seq AND e.event_name = 'message.assistant'
       AND (replace(e.payload_json, '\u0000', '\ufffd')::jsonb #>> '{metadata,context_seq}') IS NOT NULL
     ORDER BY e.event_seq DESC LIMIT 1)
  ) AS replies ORDER BY event_seq DESC LIMIT 1
),
reply AS (
  SELECT event_seq, payload_json FROM (
    (SELECT e.event_seq, e.payload_json FROM runtime.events AS e, span
     WHERE e.parent_agent_id = $1 AND NULLIF(TRIM(e.agent_id), '') IS NULL
       AND e.event_seq >= span.first_event_seq AND e.event_name = 'message.assistant'
       AND (replace(e.payload_json, '\u0000', '\ufffd')::jsonb #>> '{metadata,context_seq}') IS NOT NULL
     ORDER BY e.event_seq LIMIT 1)
    UNION ALL
    (SELECT e.event_seq, e.payload_json FROM runtime.events AS e, span
     WHERE e.agent_id = $1 AND e.event_seq >= span.first_event_seq AND e.event_name = 'message.assistant'
       AND (replace(e.payload_json, '\u0000', '\ufffd')::jsonb #>> '{metadata,context_seq}') IS NOT NULL
     ORDER BY e.event_seq LIMIT 1)
  ) AS replies ORDER BY event_seq LIMIT 1
),
next_prompt AS (
  SELECT payload_json FROM (
    (SELECT e.event_seq, e.payload_json FROM runtime.events AS e, reply
     WHERE e.parent_agent_id = $1 AND NULLIF(TRIM(e.agent_id), '') IS NULL
       AND e.event_seq > reply.event_seq AND e.event_name = 'message.user'
     ORDER BY e.event_seq LIMIT 1)
    UNION ALL
    (SELECT e.event_seq, e.payload_json FROM runtime.events AS e, reply
     WHERE e.agent_id = $1 AND e.event_seq > reply.event_seq AND e.event_name = 'message.user'
     ORDER BY e.event_seq LIMIT 1)
  ) AS prompts ORDER BY event_seq LIMIT 1
),
bounds AS (
  SELECT span.first_context_seq, span.last_context_seq,
    (SELECT (replace(payload_json, '\u0000', '\ufffd')::jsonb #>> '{metadata,context_seq}')::bigint FROM reply_before) AS reply_before_seq,
    (SELECT (replace(payload_json, '\u0000', '\ufffd')::jsonb #>> '{metadata,context_seq}')::bigint FROM reply) AS reply_seq,
    (SELECT (replace(payload_json, '\u0000', '\ufffd')::jsonb #>> '{metadata,context_seq}')::bigint FROM next_prompt) AS next_prompt_seq
  FROM span
)
SELECT least(coalesce(reply_before_seq, 0), first_context_seq) AS first_seq,
       greatest(coalesce(next_prompt_seq - 1, reply_seq, last_context_seq), last_context_seq) AS last_seq
FROM bounds
WHERE first_context_seq IS NOT NULL
