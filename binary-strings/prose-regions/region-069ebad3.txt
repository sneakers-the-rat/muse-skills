    function ensureMapLibre() {
      const version = "4.7.1";
      const scriptSelector = `script[data-maplibre-js="${version}"]`;
      const cssSelector = `link[data-maplibre-css="${version}"]`;
      const scriptStateAttribute = "data-maplibre-state";
      const scriptSrc = `https://unpkg.com/maplibre-gl@${version}/dist/maplibre-gl.js`;
      const cssHref = `https://unpkg.com/maplibre-gl@${version}/dist/maplibre-gl.css`;

      if (!document.querySelector(cssSelector)) {
        const styleLink = document.createElement("link");
        styleLink.rel = "stylesheet";
        styleLink.href = cssHref;
        styleLink.setAttribute("data-maplibre-css", version);
        document.head.appendChild(styleLink);
      }

      if (window.maplibregl && typeof window.maplibregl.Map === "function") {
        return Promise.resolve(window.maplibregl);
      }

      return new Promise((resolve, reject) => {
        function onLoadSuccess(scriptNode) {
          if (window.maplibregl && typeof window.maplibregl.Map === "function") {
            scriptNode.setAttribute(scriptStateAttribute, "loaded");
            resolve(window.maplibregl);
            return;
          }
          scriptNode.setAttribute(scriptStateAttribute, "error");
          reject(new Error("Map library load failed"));
        }

        function onLoadError(scriptNode) {
          scriptNode.setAttribute(scriptStateAttribute, "error");
          reject(new Error("Map library load failed"));
        }

        const existing = document.querySelector(scriptSelector);
        if (existing) {
          if (window.maplibregl && typeof window.maplibregl.Map === "function") {
            existing.setAttribute(scriptStateAttribute, "loaded");
            resolve(window.maplibregl);
            return;
          }
          const scriptState = existing.getAttribute(scriptStateAttribute);
          const scriptReadyState = String(existing.readyState || "").toLowerCase();
          if (scriptState === "error" || scriptReadyState === "complete") {
            existing.setAttribute(scriptStateAttribute, "error");
            reject(new Error("Map library load failed"));
            return;
          }
          existing.setAttribute(scriptStateAttribute, "loading");
          existing.addEventListener("load", () => onLoadSuccess(existing), { once: true });
          existing.addEventListener("error", () => onLoadError(existing), { once: true });
          return;
        }

        const script = document.createElement("script");
        script.src = scriptSrc;
        script.async = true;
        script.setAttribute("data-maplibre-js", version);
        script.setAttribute(scriptStateAttribute, "loading");
        script.addEventListener("load", () => onLoadSuccess(script), { once: true });
        script.addEventListener("error", () => onLoadError(script), { once: true });
        document.head.appendChild(script);
      });
    }

    function compactIndicatorStates(index, count) {
      if (count <= 4) {
        return Array.from({ length: count }, (_, itemIndex) => ({
          active: itemIndex === index,
          more: false,
        }));
      }
      if (index <= 2) {
        return [
          { active: index === 0, more: false },
          { active: index === 1, more: false },
          { active: index === 2, more: false },
          { active: false, more: true },
        ];
      }
      if (index >= count - 3) {
        return [
          { active: false, more: true },
          { active: index === count - 3, more: false },
          { active: index === count - 2, more: false },
          { active: index === count - 1, more: false },
        ];
      }
      return [
        { active: false, more: true },
        { active: false, more: false },
        { active: true, more: false },
        { active: false, more: true },
      ];
    }

      function resetGesture() {
        startX = null;
        startY = null;
        activePointerId = null;
      }

      function completeSettle() {
        if (!settleCompletion) {
          return;
        }
        // A new flick must commit the prior one before computing its target.
        if (settleTimer) {
          window.clearTimeout(settleTimer);
        }
        const completion = settleCompletion;
        settleCompletion = null;
        settleTimer = null;
        completion();
      }

      function finishGesture(event, cancelled) {
        if (startX === null || startY === null) {
          return;
        }
        if (
          activePointerId !== null &&
          event.pointerId !== undefined &&
          event.pointerId !== activePointerId
        ) {
          return;
        }
        event.preventDefault?.();
        const delta = event.clientX - startX;
        const shouldCommit =
          !cancelled &&
          activeClone &&
          Math.abs(delta) >= Math.min(42, panelWidth * 0.22);
        const direction = activeDirection;
        const transition = "transform 160ms cubic-bezier(.2,.8,.2,1)";
        selectionPanel.style.transition = transition;
        if (activeClone) {
          activeClone.style.transition = transition;
        }

        if (shouldCommit) {
          const targetIndex = activeTargetIndex;
          selectionPanel.style.transform =
            `translateX(${-direction * (panelWidth + 8)}px)`;
          activeClone.style.transform = "translateX(0)";
          settleCompletion = () => {
            selectPlace(targetIndex, {
              animate: true,
              recenter: true,
              scrollCard: false,
            });
            selectionPanel.style.transition = "none";
            selectionPanel.style.transform = "";
            clearClone();
          };
        } else {
          selectionPanel.style.transform = "";
          if (activeClone) {
            activeClone.style.transform =
              `translateX(${direction * (panelWidth + 8)}px)`;
          }
          settleCompletion = () => {
            selectionPanel.style.transition = "none";
            clearClone();
          };
        }
        settleTimer = window.setTimeout(completeSettle, 170);

        const pointerId = activePointerId;
        resetGesture();
        try {
          selectionPanel.releasePointerCapture?.(pointerId);
        } catch (_) {}
      }

      selectionPanel.addEventListener("pointerdown", (event) => {
        event.preventDefault();
        completeSettle();
        clearClone();
        selectionPanel.style.transition = "none";
        selectionPanel.style.transform = "";
        startX = event.clientX;
        startY = event.clientY;
        activePointerId = event.pointerId;
        panelWidth = selectionPanel.getBoundingClientRect().width;
        try {
          selectionPanel.setPointerCapture?.(event.pointerId);
        } catch (_) {}
      });

      selectionPanel.addEventListener("pointermove", (event) => {
        if (startX === null || startY === null) {
          return;
        }
        const deltaX = event.clientX - startX;
        const deltaY = event.clientY - startY;
        if (Math.abs(deltaX) <= Math.abs(deltaY)) {
          return;
        }
        event.preventDefault();
        const offset = Math.max(-panelWidth, Math.min(panelWidth, deltaX));
        const direction = offset < 0 ? 1 : -1;
        prepareClone(direction);
        selectionPanel.style.transform = `translateX(${offset}px)`;
        if (activeClone) {
          activeClone.style.transform =
            `translateX(${offset + direction * (panelWidth + 8)}px)`;
        }
      });

      selectionPanel.addEventListener("pointerup", (event) => {
        finishGesture(event, false);
      });
      selectionPanel.addEventListener("pointercancel", (event) => {
        finishGesture(event, true);
      });
      selectionPanel.addEventListener("lostpointercapture", (event) => {
        if (startX !== null) {
          finishGesture(event, true);
        }
      });
      selectionPanel.addEventListener("dragstart", (event) => {
        event.preventDefault();
      });
      updateCompactPageIndicator(selectedIndex);
    }

    function createCard(place, index) {
      const card = document.createElement("button");
      card.type = "button";
      card.className = "local-map-card";
      card.setAttribute("data-selected", "false");
      card.setAttribute("aria-pressed", "false");
      card.setAttribute("data-local-map-card", place.id || `local-place-${index}`);

      const name = document.createElement("span");
      name.className = "local-map-card-name";
      name.textContent = place.name;
      card.appendChild(name);

      card.addEventListener("click", () => {
        selectPlace(index, { animate: true, recenter: true, scrollCard: true });
      });

      placesList.appendChild(card);
      cards.push(card);
    }

    function renderSelectedPlaceDetails(index) {
      const place = places[index];
      if (!place) {
        selectedDetail.innerHTML = "";
        return;
      }

      selectedDetail.innerHTML = "";
      const header = document.createElement("div");
      header.className = "local-map-detail-header";

      const copy = document.createElement("div");
      copy.className = "local-map-detail-copy";

      const name = document.createElement("h4");
      name.className = "local-map-detail-title";
      name.textContent = place.name;
      copy.appendChild(name);

      const meta = document.createElement("p");
      meta.className = "local-map-card-meta";
      meta.textContent = buildMetaLine(place);
      copy.appendChild(meta);

      header.appendChild(copy);

      selectedDetail.appendChild(header);

      if (place.description) {
        const description = document.createElement("p");
        description.className = "local-map-card-description";
        description.textContent = place.description;
        selectedDetail.appendChild(description);
      }
    }

    function renderPlaceCards() {
      cards.length = 0;
      placesList.innerHTML = "";
      placesList.hidden = places.length <= 1;
      places.forEach((place, index) => {
        createCard(place, index);
      });
    }

    function applySelection(index) {
      markers.forEach((entry, markerIndex) => {
        const selected = markerIndex === index;
        entry.element.setAttribute("data-selected", selected ? "true" : "false");
        entry.element.innerHTML = markerSvg(markerIndex, selected);
      });
      cards.forEach((card, cardIndex) => {
        const selected = cardIndex === index;
        card.setAttribute("data-selected", selected ? "true" : "false");
        card.setAttribute("aria-pressed", selected ? "true" : "false");
      });
      renderSelectedPlaceDetails(index);
      updateCompactPageIndicator(index);
    }

    function selectPlace(index, options = {}) {
      if (!places[index]) {
        return;
      }

      const { animate = true, recenter = true, scrollCard = true } = options;
      selectedIndex = index;
      applySelection(index);

      const place = places[index];
      const center = [place.longitude, place.latitude];

      if (map && recenter) {
        const move = { center, zoom: flyZoom, essential: true };
        if (animate) {
          map.flyTo(move);
        } else {
          map.jumpTo(move);
        }
      }

      if (scrollCard && cards[index]) {
        cards[index].scrollIntoView({
          behavior: "smooth",
          block: "nearest",
          inline: "center",
        });
      }
    }

    function initializeMap() {
      const center = [places[0].longitude, places[0].latitude];
      renderPlaceCards();
      setupCompactCarousel();
      selectPlace(0, { animate: false, recenter: false, scrollCard: false });

      const style = String(activeStyleUrl() || "").trim();
      if (!style) {
        showError("Map style URL is missing.");
        return;
      }

      bindZoomControls();
      attemptMapInitialization(center, style);
    }

    function attemptMapInitialization(centerOverride, style) {
      const center = Array.isArray(centerOverride)
        ? centerOverride
        : [places[0].longitude, places[0].latitude];
      clearErrors();
      destroyMapInstance();

      map = new window.maplibregl.Map({
        attributionControl: false,
        center,
        container: mapCanvas,
        maplibreLogo: false,
        scrollZoom: false,
        style,
        zoom: flyZoom,
      });

      const isMobileLayout = window.matchMedia("(max-width: 980px)").matches;
      const fitBoundsPadding = isMobileLayout
        ? { top: 40, right: 24, bottom: 40, left: 24 }
        : { top: 40, right: 40, bottom: 40, left: 40 };

      startStyleLoadWatchdog();

      map.on("style.load", () => {
        styleLoaded = true;
        clearLoadWatchdog();
      });

      map.on("error", (event) => {
        if (!styleLoaded && event && event.error && event.error.message) {
          mapFailureReason = String(event.error.message);
        }
      });

      map.on("load", () => {
        mapLoaded = true;
        clearLoadWatchdog();

        places.forEach((place, index) => {
          const markerElement = document.createElement("button");
          markerElement.type = "button";
          markerElement.className = "local-map-pin";
          markerElement.setAttribute("data-selected", "false");
          markerElement.setAttribute("aria-label", place.name);
          markerElement.innerHTML = markerSvg(index, false);
          markerElement.addEventListener("click", () => {
            selectPlace(index, { animate: true, recenter: true, scrollCard: true });
          });

          const marker = new window.maplibregl.Marker({
            element: markerElement,
            anchor: "bottom",
          })
            .setLngLat([place.longitude, place.latitude])
            .addTo(map);

          markers.push({ marker, element: markerElement });
        });

        if (places.length > 1) {
          const points = places.map((place) => [place.longitude, place.latitude]);
          const bounds = new window.maplibregl.LngLatBounds(points[0], points[1]);
          for (let index = 2; index < points.length; index += 1) {
            bounds.extend(points[index]);
          }
          map.fitBounds(bounds, {
            duration: 0,
            padding: fitBoundsPadding,
          });
          selectPlace(selectedIndex, { animate: false, recenter: false, scrollCard: false });
        } else {
          selectPlace(selectedIndex, { animate: false, recenter: true, scrollCard: false });
        }
      });
    }

    ensureMapLibre()
      .then(() => {
        initializeMap();
      })
      .catch(() => {
        showError("Map could not load. Check map style and network access.");
      });

    root.addEventListener("keydown", (event) => {
      if (event.key === "ArrowDown") {
        const next = (selectedIndex + 1) % places.length;
        selectPlace(next, { animate: true, recenter: true, scrollCard: true });
      }
      if (event.key === "ArrowUp") {
        const next = (selectedIndex - 1 + places.length) % places.length;
        selectPlace(next, { animate: true, recenter: true, scrollCard: true });
      }
    });
  })();
