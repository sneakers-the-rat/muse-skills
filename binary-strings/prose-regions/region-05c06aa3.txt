
WITH event_transcript_rows AS (
  SELECT
    e.event_name,
    e.visibility::text AS visibility,
    coalesce(e.role::text, m.role::text, 'system') AS role,
    e.source,
    e.transcript_surface::text AS transcript_surface,
    e.parent_message_id,
    e.parent_agent_id,
    coalesce(e.message_id, m.message_id, '') AS message_id
  FROM runtime.events e
  LEFT JOIN runtime.messages m ON m.event_seq = e.event_seq
)
SELECT LEAST(COUNT(DISTINCT e.message_id), $1)::bigint
FROM event_transcript_rows AS e
WHERE À%
  AND e.event_name = 'message.user'
