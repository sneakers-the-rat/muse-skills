SELECT bool_and(coalesce(i.indisvalid, false)) FROM unnest($1::text[]) AS n(name)
                 LEFT JOIN pg_index i ON i.indexrelid=to_regclass(n.name)
JOIN LATERAL (
  SELECT MAX(COALESCE(f.text_value, f.scalar_value))
           FILTER (WHERE f.field_path = '/metadata/kind') AS kind,
         MAX(COALESCE(f.text_value, f.scalar_value))
           FILTER (WHERE f.field_path = '/metadata/action') AS action,
         MAX(COALESCE(f.text_value, f.scalar_value))
           FILTER (WHERE f.field_path = '/metadata/handoff_message_id') AS handoff_message_id,
         MAX(COALESCE(f.text_value, f.scalar_value))
           FILTER (WHERE f.field_path = '/metadata/handoff_source') AS handoff_source,
         MAX(COALESCE(f.text_value, f.scalar_value))
           FILTER (WHERE f.field_path = '/metadata/submission_message_id') AS submission_message_id,
         MAX(COALESCE(f.text_value, f.scalar_value))
           FILTER (WHERE f.field_path = '/metadata/submission_source') AS submission_source
  FROM runtime.event_payload_fields AS f
  WHERE f.event_seq = e.event_seq AND f.field_path IN (
    '/metadata/kind', '/metadata/action', '/metadata/handoff_message_id',
    '/metadata/handoff_source', '/metadata/submission_message_id', '/metadata/submission_source'
  )
) AS metadata ON true
