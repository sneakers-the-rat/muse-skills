WITH repair AS (
  SELECT *
  FROM unnest(
    $2::bigint[],
    $3::boolean[],
    $4::text[],
    $5::boolean[],
    $6::text[],
    $7::boolean[],
    $8::text[]
  ) AS input(
    seq,
    client_context_requested,
    client_context_json,
    reply_prefix_requested,
    reply_prefix,
    provenance_requested,
    provenance_json
  )
),
locked AS MATERIALIZED (
  SELECT projection.agent_id, projection.seq
  FROM context_item_resume_projection AS projection
  JOIN repair ON repair.seq = projection.seq
  WHERE projection.agent_id = $1
  ORDER BY projection.seq ASC
  FOR UPDATE OF projection
)
UPDATE context_item_resume_projection AS projection
SET client_context_json = CASE
      WHEN repair.client_context_requested THEN repair.client_context_json
      ELSE projection.client_context_json
    END,
    client_context_projected =
      COALESCE(projection.client_context_projected, FALSE)
      OR repair.client_context_requested,
    reply_prefix = CASE
      WHEN repair.reply_prefix_requested THEN repair.reply_prefix
      ELSE projection.reply_prefix
    END,
    reply_prefix_projected =
      COALESCE(projection.reply_prefix_projected, FALSE)
      OR repair.reply_prefix_requested,
    provenance_json = CASE
      WHEN repair.provenance_requested THEN repair.provenance_json
      ELSE projection.provenance_json
    END,
    provenance_projected =
      COALESCE(projection.provenance_projected, FALSE)
      OR repair.provenance_requested,
    updated_at = now()
FROM repair
JOIN locked
  ON locked.seq = repair.seq
WHERE projection.agent_id = locked.agent_id
  AND projection.seq = locked.seq
