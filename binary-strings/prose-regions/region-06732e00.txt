       payload_json, created_at, updated_at,
       ARRAY(
         SELECT json_build_object(
           'accepted_ordinal', mailbox.accepted_ordinal,
           'state', mailbox.state,
           'checkpoint_custody', mailbox.checkpoint_custody,
           'submission_json', mailbox.submission_json
         )::text
         FROM agent.message_mailbox mailbox

       ) AS mailbox_overlay_rows
FROM runtime_restart_checkpoints
WHERE agent_id > $1 AND ($2::text IS NULL OR agent_id <> $2)
ORDER BY agent_id ASC
LIMIT $3
