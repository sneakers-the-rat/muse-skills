WITH boundary AS MATERIALIZED (
  SELECT GREATEST(
    COALESCE((SELECT prompt_floor_seq FROM agents WHERE id = $1), -1),
    COALESCE((SELECT checkpoint_seq FROM agent_compactions
              WHERE agent_id = $1 AND has_replacement_history
              ORDER BY id DESC LIMIT 1), -1)
  ) AS floor_seq
),
-- Provenance is a consumed-identity fence. Validate through migration 399's
-- expression index before deleting anything; treating malformed typed data as
-- absent would let a quarantined public voice item replay on a later import.
validation AS MATERIALIZED (
  SELECT EXISTS (
    SELECT 1
    FROM context_item_resume_projection invalid
    CROSS JOIN boundary
    WHERE invalid.agent_id = $1
      AND NOT (invalid.seq = ANY($2))
      AND invalid.seq > boundary.floor_seq
      AND invalid.provenance_json IS NOT NULL
      AND md5(agent.resume_projection_message_id_v1(
          invalid.provenance_json
      ))::uuid IS NULL
  ) AS invalid_provenance
  FROM boundary
),
-- Carry the statement-snapshot row version into DELETE. A concurrent event
-- writer changes ctid, so EvalPlanQual retains that projection rather
-- than deleting an identity whose newly committed event is not in our snapshot.
rewrite_scope AS MATERIALIZED (
  SELECT projection.ctid AS row_ctid,
         agent.resume_projection_message_id_v1(
           projection.provenance_json
         ) AS message_id
  FROM context_item_resume_projection projection
  CROSS JOIN boundary
  WHERE (SELECT NOT invalid_provenance FROM validation)
    AND projection.agent_id = $1
    AND NOT (projection.seq = ANY($2))
    AND projection.seq > boundary.floor_seq
),
event_scope AS NOT MATERIALIZED (
  SELECT candidate.row_ctid, candidate.message_id
  FROM rewrite_scope candidate
  WHERE candidate.message_id IS NOT NULL
),
scope_stats AS MATERIALIZED (
  SELECT count(*) FILTER (WHERE candidate.message_id IS NOT NULL) AS event_candidate_count,
         count(*) FILTER (WHERE candidate.message_id IS NULL) AS null_candidate_count
  FROM rewrite_scope candidate
),
-- Bound random event-index probes for small rewrites. For genuinely large
-- rewrites, leave the anti-join flattenable so PostgreSQL can hash events once.
deletable AS (
  SELECT branch.row_ctid
  FROM scope_stats scope
  CROSS JOIN LATERAL (
    SELECT candidate.row_ctid
    FROM rewrite_scope candidate
    WHERE scope.null_candidate_count > 0
      AND candidate.message_id IS NULL
    OFFSET 0
  ) branch
  UNION ALL
  SELECT branch.row_ctid
  FROM scope_stats scope
  CROSS JOIN LATERAL (
    SELECT candidate.row_ctid
    FROM event_scope candidate
    WHERE scope.event_candidate_count BETWEEN 1 AND 4096
      AND NOT EXISTS (
        SELECT 1 FROM runtime.events e
        WHERE e.message_id = candidate.message_id
          AND COALESCE(NULLIF(TRIM(e.agent_id), ''), e.parent_agent_id) <> $1
          AND e.transcript_surface = 'main_chat'
          AND (e.visibility = 'public'
            OR (e.visibility = 'internal'
              AND e.event_name = 'message.internal'
              AND e.source = 'runtime.voice_companion_context'
              AND NOT EXISTS (
                SELECT 1 FROM runtime.event_payload_fields role_field
                WHERE role_field.event_seq = e.event_seq
                  AND role_field.field_path = '/role'
                  AND role_field.scalar_type = 'string'
                  AND COALESCE(
                    role_field.text_value, role_field.scalar_value
                  ) <> 'developer'
              )))
        OFFSET 0
      )
    OFFSET 0
  ) branch
  UNION ALL
  SELECT branch.row_ctid
  FROM scope_stats scope
  CROSS JOIN LATERAL (
    SELECT candidate.row_ctid
    FROM event_scope candidate
    WHERE scope.event_candidate_count > 4096
      AND NOT EXISTS (
        SELECT 1 FROM runtime.events e
        WHERE e.message_id = candidate.message_id
          AND COALESCE(NULLIF(TRIM(e.agent_id), ''), e.parent_agent_id) <> $1
          AND e.transcript_surface = 'main_chat'
          AND (e.visibility = 'public'
            OR (e.visibility = 'internal'
              AND e.event_name = 'message.internal'
              AND e.source = 'runtime.voice_companion_context'
              AND NOT EXISTS (
                SELECT 1 FROM runtime.event_payload_fields role_field
                WHERE role_field.event_seq = e.event_seq
                  AND role_field.field_path = '/role'
                  AND role_field.scalar_type = 'string'
                  AND COALESCE(
                    role_field.text_value, role_field.scalar_value
                  ) <> 'developer'
              )))
      )
    OFFSET 0
  ) branch
),
deleted AS (
  DELETE FROM context_item_resume_projection projection
  WHERE projection.agent_id = $1
    AND projection.ctid = ANY(ARRAY(
      SELECT candidate.row_ctid
      FROM deletable candidate
    ))
)

  SELECT COALESCE(prompt_floor_seq, -1) AS prompt_floor_seq
  FROM agents
  WHERE id = $1

summary AS (
  SELECT COUNT(*) > $3 AS has_earlier_items,
         CASE
           WHEN (SELECT prompt_floor_seq FROM current_floor) IS DISTINCT FROM $4 THEN FALSE
           WHEN $2 = $4 + 1 THEN COUNT(*) = 0
           ELSE $2 > $4 + 1
             AND COUNT(*) = $2 - $4 - 1
             AND MIN(seq) = $4 + 1
             AND MAX(seq) = $2 - 1
         END AS sequence_complete
  FROM context_items
  WHERE agent_id = $1 AND seq > $4 AND seq < $2
),
bounded AS (
  SELECT ci.context_item_id,
         ci.seq,
         (extract(epoch from ci.created_at))::bigint AS created_at,
         COALESCE(p.message_source, ci.message_source) AS message_source,
         ci.item_json AS item_json,

         p.provenance_projected

    ON p.agent_id = ci.agent_id AND p.seq = ci.seq
  WHERE ci.agent_id = $1 AND ci.seq > $4 AND ci.seq < $2
