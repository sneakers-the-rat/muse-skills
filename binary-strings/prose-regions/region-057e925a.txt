update runtime.browser_tasks
                    set deadline_at = greatest(deadline_at, now() + interval '15 minutes'),
                        updated_at = now()
                    where task_id = $1 and admission_seq is not null
                      and owner_agent_id is not distinct from $2
                      and stream_owner_message_id = $3 and state_revision = $4
                      and status = 'user_controlled' and deadline_at > now()
                      and completed_at is null
