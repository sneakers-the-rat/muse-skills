WITH link_scope AS MATERIALIZED (
  SELECT p.seq, p.provenance_json IS NOT NULL AS has_provenance,
         agent.resume_projection_message_id_v1(p.provenance_json) AS message_id
  FROM agent.context_item_resume_projection p
  WHERE p.agent_id = $1 AND p.seq = ANY($2)
),
validation AS MATERIALIZED (
  SELECT EXISTS (
    SELECT 1 FROM link_scope
    WHERE has_provenance AND message_id IS NULL
  ) AS invalid_provenance
),
linked AS (
  UPDATE agent.context_items ci
  SET message_id = m.message_id
  FROM link_scope p
  JOIN runtime.messages m ON m.message_id = p.message_id
  CROSS JOIN validation
  WHERE NOT validation.invalid_provenance
    AND ci.agent_id = $1 AND ci.seq = p.seq
    AND starts_with(m.message_id, 'voice-tool-context-')
    AND ci.item_kind IN ('message', 'message_text')
    AND ci.message_id IS DISTINCT FROM m.message_id
)
