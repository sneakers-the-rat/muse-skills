FROM unnest($2::text[], $3::bigint[], $4::bigint[], $5::timestamptz[], $6::timestamptz[],
            $7::text[])

  generation = excluded.generation,
  deletion_seq = greatest(d.deletion_seq, excluded.deletion_seq),
  deleted_at_ms = CASE WHEN excluded.deletion_seq >= d.deletion_seq
                       THEN coalesce(excluded.deleted_at_ms, d.deleted_at_ms)
                       ELSE d.deleted_at_ms END,
  first_checked_at = CASE WHEN excluded.deletion_seq > d.deletion_seq
                            THEN excluded.first_checked_at
                          WHEN excluded.deletion_seq = d.deletion_seq
                            THEN least(d.first_checked_at, excluded.first_checked_at)
                          ELSE d.first_checked_at END,
  last_checked_at = CASE WHEN excluded.deletion_seq >= d.deletion_seq
                         THEN excluded.last_checked_at
                         ELSE d.last_checked_at END,
  last_state = CASE WHEN excluded.deletion_seq >= d.deletion_seq
                    THEN excluded.last_state
                    ELSE d.last_state END
