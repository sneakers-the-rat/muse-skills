
  AND e.event_name IN ('message.user', 'message.assistant')
  AND (e.parent_message_id IS NULL OR trim(e.parent_message_id) = '')
  AND ($2 IS NULL OR e.message_id <> $2)
  AND ($2 IS NULL OR e.reply_to_message_id IS NULL OR e.reply_to_message_id <> $2)
  AND (
        $3 IS NULL OR
        COALESCE(NULLIF(COALESCE(context_seq_field.text_value, context_seq_field.scalar_value), '')::bigint, -1) < $3
      )
ORDER BY e.event_seq DESC
LIMIT 1
