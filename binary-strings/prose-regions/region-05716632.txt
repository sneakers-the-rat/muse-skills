SELECT items_json, revision, updated_at_ms, continuation_key,
       continuation_generation, continuation_reminders
FROM runtime.agent_todo_snapshots
WHERE conversation_id = $1 AND agent_identity = $2

UPDATE runtime.agent_todo_snapshots
SET continuation_reminders = $5
WHERE conversation_id = $1 AND agent_identity = $2
  AND continuation_key = $3 AND continuation_generation = $4
  AND continuation_reminders < $5

INSERT INTO runtime.agent_todo_snapshots
  (conversation_id, agent_identity, items_json, revision, updated_at_ms,
   continuation_key, continuation_generation, continuation_reminders)
VALUES ($1, $2, $3, 1, (extract(epoch from now()) * 1000)::bigint, $4, 1, 0)
ON CONFLICT (conversation_id, agent_identity) DO UPDATE SET
  items_json = excluded.items_json,
  revision = agent_todo_snapshots.revision + 1,
  updated_at_ms = excluded.updated_at_ms,
  continuation_key = excluded.continuation_key,
  continuation_generation = CASE
    WHEN agent_todo_snapshots.continuation_key = excluded.continuation_key
    THEN agent_todo_snapshots.continuation_generation
    ELSE agent_todo_snapshots.continuation_generation + 1 END,
  continuation_reminders = CASE
    WHEN agent_todo_snapshots.continuation_key = excluded.continuation_key
    THEN agent_todo_snapshots.continuation_reminders ELSE 0 END
RETURNING revision

UPDATE runtime.agent_todo_snapshots
SET continuation_key = $4,
    continuation_generation = CASE WHEN continuation_key IS DISTINCT FROM $4
        THEN continuation_generation + 1 ELSE continuation_generation END,
    continuation_reminders = CASE WHEN continuation_key IS DISTINCT FROM $4
        THEN 0 ELSE continuation_reminders END
WHERE conversation_id = $1 AND agent_identity = $2 AND revision = $3
RETURNING continuation_key, continuation_generation, continuation_reminders
