        select
            s.ord,
            c.external_contact_id as contact_id,
            coalesce(c.display_name, '') as display_name,
            coalesce(
              array(
                select p.phone_raw
                from device.contact_phones p
                where p.contact_id = c.contact_id
                order by p.contact_phone_id
              ),
              array[]::text[]
            ) as phones,
            coalesce(
              array(
                select p.phone_e164
                from device.contact_phones p
                where p.contact_id = c.contact_id
                order by p.contact_phone_id
              ),
              array[]::text[]
            ) as phones_e164
        from unnest($2::text[]) with ordinality as s(digits, ord)
        cross join lateral (
            select c.contact_id, c.external_contact_id, c.display_name
            from device.contacts c
            join device.nodes n
              on n.node_id = c.node_id
            where s.digits <> ''
              and n.pairing_status <> 'revoked'
              and c.deleted_at is null
              and c.node_id = $1
              and exists (
                select 1 from device.contact_phones p
                where p.contact_id = c.contact_id
                  and (strpos(regexp_replace(p.phone_raw, '[^0-9]', '', 'g'), s.digits) > 0
                    or strpos(regexp_replace(coalesce(p.phone_e164, ''), '[^0-9]', '', 'g'), s.digits) > 0)
              )
            order by lower(coalesce(c.display_name, c.external_contact_id)), lower(c.external_contact_id)
            limit $3
        ) c
