SELECT
  e.event_seq AS seq,
  (extract(epoch from e.created_at) * 1000)::bigint AS occurred_at_ms,
  e.event_name,
  e.visibility::text AS visibility,
  coalesce(e.role::text, m.role::text, 'system') AS role,
  e.source,
  e.stream_lane,
  e.transcript_surface::text AS transcript_surface,
  e.request_id,
  e.root_request_id,
  e.parent_request_id,
  coalesce(e.message_id, m.message_id, '') AS message_id,
  e.reply_to_message_id,
  e.reply_target_message_id,
  e.agent_id,
  e.parent_agent_id,
  e.parent_message_id,
  case
    when derived_backlog.event_seq is not null
    then coalesce(e.payload_json, '{}'::text)
    else '{}'::text
  end AS payload_json,
  CASE
    WHEN e.event_name = 'delta.presentation' THEN NULLIF(m.body, '')
    ELSE m.body
  END AS display_text,
  case when e.display_text_ready then 1::bigint else 0::bigint end AS display_text_ready,
  null::bigint AS display_text_updated_at_ms,
  coalesce(e.idempotency_key, e.event_id::text) AS idempotency_key,
  channel_ctx.chat_context_json,
  coalesce(
    nullif(channel_ctx.chat_context_json::jsonb ->> 'origin_provider', ''),
    nullif(channel_ctx.chat_context_json::jsonb ->> 'provider', ''),
    external_transport.provider
  ) AS origin_provider,
  e.chat_kind,
  mr.reaction_emoji
FROM events e

LEFT JOIN messages m ON m.event_seq = e.event_seq
LEFT JOIN chat_event_derived_write_backlog derived_backlog ON derived_backlog.event_seq = e.event_seq
LEFT JOIN message_reactions mr ON mr.message_id = coalesce(e.message_id, m.message_id)
LEFT JOIN LATERAL (
  SELECT coalesce(
    (
      SELECT coalesce(ctx_field.text_value, ctx_field.scalar_value)
      FROM event_payload_fields ctx_field
      WHERE ctx_field.event_seq = e.event_seq
        AND ctx_field.field_path = '/__hatch/chat_context_json'
      LIMIT 1
    ),
    e.chat_context_json,
    nullif((e.payload_json::jsonb -> 'chat_context')::text, 'null')
  ) AS chat_context_json
) channel_ctx ON true
LEFT JOIN LATERAL (
  SELECT c.provider
  FROM chat.event_transports c
  WHERE c.event_seq = e.event_seq
    AND nullif(c.provider, '') IS NOT NULL
    AND c.provider <> 'main'
  ORDER BY c.provider ASC
  LIMIT 1
) external_transport ON true

SELECT requested.seq AS context_seq,
       projection.client_context_json,
       COALESCE(projection.client_context_projected, FALSE) AS client_context_projected,
       fallback.event_name,
       fallback.source,
       fallback.payload_json
FROM unnest($2::bigint[]) AS requested(seq)
LEFT JOIN agent.context_item_resume_projection AS projection
  ON projection.agent_id = $1
 AND projection.seq = requested.seq
 AND $3
LEFT JOIN LATERAL (
  SELECT e.event_name, e.source, e.payload_json
  FROM runtime.events AS e
  WHERE projection.client_context_projected IS NOT TRUE
    AND COALESCE(NULLIF(TRIM(e.agent_id), ''), e.parent_agent_id) = $1
    AND (replace(e.payload_json, '\u0000', '\ufffd')::jsonb #>> '{metadata,context_seq}') = requested.seq::text
    AND e.event_name IN ('message.user', 'message.internal')
    AND e.payload_json IS NOT NULL
  ORDER BY e.event_seq DESC
  LIMIT 1
) AS fallback ON TRUE
ORDER BY requested.seq DESC

WITH event_rows AS (
SELECT e.reply_to_message_id,
       e.message_id,
       e.event_name,
       e.source,
       e.role::text AS role,
       e.visibility::text AS visibility,
       e.request_id,
       e.root_request_id,
       e.parent_request_id,
       CASE
         WHEN e.message_id LIKE 'assistant-msg-%'
         THEN substring(e.message_id FROM char_length('assistant-msg-') + 1)
         ELSE NULL
       END AS message_execution_id,
       NULL::text AS root_message_execution_id,
       (replace(e.payload_json, '\u0000', '\ufffd')::jsonb #>> '{metadata,context_seq}')::bigint AS context_seq,
       e.event_seq
FROM runtime.events AS e
WHERE COALESCE(NULLIF(TRIM(e.agent_id), ''), e.parent_agent_id) = $1
  AND e.event_name IN ('message.user', 'message.assistant', 'message.internal')
  AND (replace(e.payload_json, '\u0000', '\ufffd')::jsonb #>> '{metadata,context_seq}') = ANY($2)
)
SELECT event_rows.*, NULL::text AS imported_provenance_json
FROM event_rows
UNION ALL
SELECT NULL, NULL, NULL, NULL, NULL, NULL, NULL, NULL, NULL, NULL, NULL,
       projection.seq, NULL, projection.provenance_json
FROM agent.context_item_resume_projection AS projection
WHERE projection.agent_id = $1
  AND projection.seq = ANY($3)
  AND projection.provenance_projected
  AND NOT EXISTS (SELECT 1 FROM event_rows WHERE context_seq = projection.seq)
ORDER BY context_seq DESC, event_seq DESC

SELECT e.source, ci.seq AS context_seq
FROM agent.context_items AS ci
JOIN runtime.event_payload_fields AS ctx ON ctx.field_path = '/metadata/context_seq'
  AND ctx.scalar_value = ci.seq::text
JOIN runtime.events AS e ON e.event_seq = ctx.event_seq
  AND COALESCE(NULLIF(TRIM(e.agent_id), ''), e.parent_agent_id) = ci.agent_id
WHERE ci.agent_id = $1
  AND e.event_name IN ('message.user', 'message.assistant', 'message.internal')
  AND ci.seq = ANY($2)
ORDER BY ci.seq DESC, e.event_seq DESC

SELECT fallback.reply_to_message_id, requested.seq AS context_seq,
       fallback.payload_json, fallback.event_seq,
       CASE WHEN projection.reply_prefix_projected
            THEN projection.reply_prefix END AS imported_reply_prefix
FROM (SELECT DISTINCT unnest($2::bigint[]) AS seq) AS requested
LEFT JOIN agent.context_item_resume_projection AS projection
  ON projection.agent_id = $1 AND projection.seq = requested.seq
LEFT JOIN LATERAL (
    SELECT e.reply_to_message_id, e.payload_json, e.event_seq
    FROM runtime.events AS e
    WHERE projection.reply_prefix_projected IS NOT TRUE
      AND COALESCE(NULLIF(TRIM(e.agent_id), ''), e.parent_agent_id) = $1
      AND (replace(e.payload_json, '\u0000', '\ufffd')::jsonb #>> '{metadata,context_seq}') = requested.seq::text

      AND e.role = 'user'
      AND e.payload_json IS NOT NULL
      AND EXISTS (
          SELECT 1 FROM agent.context_items ci
          WHERE ci.agent_id = $1 AND ci.seq = requested.seq
      )
    ORDER BY e.event_seq DESC
) AS fallback ON TRUE
ORDER BY requested.seq DESC, fallback.event_seq DESC
