select task_id, title,
  case
    when status in ('queued', 'starting', 'running') then 'working'
    when status = 'user_controlled' then 'user_control'
    when status = 'parked_outcome' and owner_agent_id is null then 'idle'
    when status = 'parked_outcome' then coalesce(outcome_status, 'waiting')
    else status
  end as state,
  coalesce(owner_kind = 'user', false) as started_by_user,
  latest_tab_json as current_page,
  profile_id,
  status,
  broker_instance,
  node_id,
  root_session_id,
  owner_kind,
  request_trace_context_json,
  chat_context_json
from (
  select * from runtime.browser_tasks
  where root_session_id = $1
    and ($2::text is null or task_id > $2)
    and owner_kind is distinct from 'research'
  union all
  select * from runtime.browser_tasks
  where $3
    and owner_kind = 'cron' and broker_instance <> 'proactivity'
    and root_session_id <> $1
    and ($2::text is null or task_id > $2)
) as task
where status <> 'superseded'
order by task_id
