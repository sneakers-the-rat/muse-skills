
SELECT created_at
FROM runtime.events
WHERE transcript_surface = 'main_chat'
  AND visibility = 'public'
  AND role = 'user'
  AND event_name = 'message.user'
  AND source NOT IN (
    'runtime.action',
    'runtime.audio_transcript',
    'runtime.avatar_backfill',
    'runtime.cancellation',
    'runtime.device_sync',
    'runtime.device_sync_handoff',
    'runtime.external_sync',
    'runtime.filesystem',
    'runtime.media_upload',
    'runtime.monitoring',
    'runtime.onboarding',
    'runtime.reaction',
    'runtime.skill_invalidation',
    'runtime.subagent_progress',
    'scheduler.cron',
    'security.safety',
    'security.sentinel'
  )
ORDER BY event_seq DESC
