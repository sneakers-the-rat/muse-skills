  e.event_seq,
  e.event_name,
  (extract(epoch from e.created_at) * 1000)::bigint AS occurred_at_ms,
  e.payload_json,
  CASE
    WHEN e.event_name = 'delta.presentation' THEN NULLIF(m.body, '')
    ELSE m.body
  END AS display_text,
  e.chat_context_json,
  b.resources_json
FROM chat_event_derived_write_backlog b
JOIN events e ON e.event_seq = b.event_seq
LEFT JOIN messages m ON m.event_seq = e.event_seq
WHERE b.deadlettered_at IS NULL
  AND b.attempts < $2
  AND e.payload_json IS NOT NULL
ORDER BY b.created_at ASC, b.event_seq ASC
