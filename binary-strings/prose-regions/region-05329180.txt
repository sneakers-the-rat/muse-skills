SELECT
  preferences.active_root_id,
  agents.id AS agent_id,
  agents.kind,
  NULLIF(trim(agents.session_id), '') AS session_id,
  agents.model
FROM chat_preferences AS preferences
LEFT JOIN agents AS agents
  ON agents.id = preferences.active_root_id
WHERE preferences.id = $1
WITH public_root AS MATERIALIZED (
    SELECT id, COALESCE(prompt_floor_seq, -1) AS prompt_floor_seq
    FROM agents
    WHERE id = $1
),
-- Root rows above a call agent's seed bound, each with the event that wrote it.
late_writes AS MATERIALIZED (
    SELECT e.event_seq, e.writer
    FROM context_item_resume_projection late
    CROSS JOIN LATERAL (
        SELECT e.event_seq,
               COALESCE(NULLIF(TRIM(e.agent_id), ''), e.parent_agent_id) AS writer
        FROM runtime.events e
        WHERE e.message_id = agent.resume_projection_message_id_v1(
                late.provenance_json
            )
          AND e.parent_agent_id = $1
        OFFSET 0
    ) e
    WHERE $4::bigint IS NOT NULL
      AND late.agent_id = $1
      AND late.seq > $4
      -- Always true. With the bound above it reads as a narrow range, so a
      -- generic plan reads a large root by key range, not through the
      -- agent-only digest index.
      AND late.seq <= 9223372036854775807
      AND late.context_item_id IS NOT NULL
      AND late.provenance_json IS NOT NULL
      -- The root writes a call's voice S2S record itself; no call row backs it.
      AND NOT COALESCE(starts_with(late.message_source, 'runtime.voice_s2s_record.'), false)
),
producers AS MATERIALIZED (
    SELECT id, GREATEST(prompt_floor_seq, COALESCE($4::bigint, -1)) AS after_seq,
           true AS is_public_root
    FROM public_root
    WHERE id <> $3
    UNION ALL
    -- Call agents are found from their ids, one key lookup each: a root keeps
    -- every call agent it ever had, and no index covers
    -- presentation_root_session_id. Only owners of pending rows and late
    -- writers can contribute rows below.
    SELECT producer.id, producer.after_seq, false AS is_public_root
    FROM public_root
    CROSS JOIN (
        SELECT pending.agent_id
        FROM (
            SELECT DISTINCT p.agent_id
            FROM context_item_resume_projection p
            WHERE p.provenance_json IS NOT NULL
              AND p.public_import_exempt IS NOT TRUE
              AND p.public_import_done_provenance_md5
                  IS DISTINCT FROM md5(p.provenance_json)::uuid
            OFFSET 0
        ) pending
        UNION
        SELECT writer FROM late_writes
    ) candidate
    CROSS JOIN LATERAL (
        SELECT a.id,
               CASE WHEN pg_input_is_valid(
                   a.spawn_metadata_json::jsonb #>> '{realtime_voice,root_seed_context_seq}',
                   'bigint'
               ) THEN GREATEST(
                   (a.spawn_metadata_json::jsonb #>> '{realtime_voice,root_seed_context_seq}')::bigint,
                   -1
               ) ELSE -1 END AS after_seq
        FROM agents a
        WHERE a.id = candidate.agent_id
          AND a.presentation_root_session_id = public_root.id
          AND a.id <> public_root.id AND a.id <> $3
          AND a.kind = 'root' AND a.depth = 0
          AND a.agent_type = 'realtime_voice'
          AND a.spawn_metadata_json::jsonb #> '{realtime_voice,prompt_floor_seq}'
              = to_jsonb(public_root.prompt_floor_seq)
        OFFSET 0
    ) producer
),
-- A call agent's seed holds the root only through its bound. Root rows above
-- the bound that copy a call's message lead back to that call's row, which the
-- root's stamp would otherwise hide from this target.
root_late_context AS MATERIALIZED (
    SELECT call_row.agent_id, call_row.seq, call_row.message_id
    FROM late_writes e
    JOIN producers
      ON producers.id = e.writer AND NOT producers.is_public_root
    JOIN runtime.event_payload_fields context_seq
      ON context_seq.event_seq = e.event_seq
     AND context_seq.field_path = '/metadata/context_seq'
    -- One key probe per late row. As a join, a low caller estimate lets
    -- PostgreSQL hash every call row since the last clear instead.
    CROSS JOIN LATERAL (
        SELECT p.agent_id, p.seq,
               agent.resume_projection_message_id_v1(p.provenance_json) AS message_id
        FROM context_item_resume_projection p
        WHERE p.agent_id = producers.id
          AND p.seq = CASE WHEN pg_input_is_valid(
                  COALESCE(context_seq.text_value, context_seq.scalar_value), 'bigint'
              ) THEN COALESCE(context_seq.text_value, context_seq.scalar_value)::bigint END
          AND p.seq > producers.after_seq
          AND p.context_item_id IS NOT NULL
          AND p.provenance_json IS NOT NULL
        OFFSET 0
    ) call_row
),
-- Stored JSON may contain escaped NUL. Normalize active JSON escapes while
-- preserving literal backslash sequences; keep the durable text unchanged.
producer_context AS MATERIALIZED (
    SELECT candidate.agent_id, candidate.seq, candidate.message_id
    FROM producers
    -- Fence one range probe per producer. Without the lateral boundary,
    -- PostgreSQL can hash-join producers against a full projection-table scan.
    CROSS JOIN LATERAL (
        SELECT p.agent_id, p.seq,
               agent.resume_projection_message_id_v1(
                   p.provenance_json
               ) AS message_id
        FROM context_item_resume_projection p
        WHERE producers.is_public_root
          AND p.agent_id = producers.id
          AND p.seq > producers.after_seq
          -- Implied by the two lines above, but visible to the planner: a
          -- materialized CTE hides the root id and bound, so it otherwise
          -- assumes an average agent's few rows and can read a large root's
          -- whole history through the agent-only digest index. The always-true
          -- upper bound covers generic plans, as in `late_writes`.
          AND p.agent_id = $1
          AND p.seq > COALESCE($4::bigint, -1)
          AND p.seq <= 9223372036854775807
          AND p.context_item_id IS NOT NULL
          AND p.provenance_json IS NOT NULL
          -- A call's voice S2S record is voice-only and never imported.
          AND NOT COALESCE(starts_with(p.message_source, 'runtime.voice_s2s_record.'), false)
        UNION ALL
        -- Call rows the root already holds are skipped; a call that started
        -- before the root took one gets it from `root_late_context` below.
        -- Keep the last three conditions identical to migration 0402's
        -- partial index predicate.
        SELECT p.agent_id, p.seq,
               agent.resume_projection_message_id_v1(
                   p.provenance_json
               ) AS message_id
        FROM context_item_resume_projection p
        WHERE NOT producers.is_public_root
          AND p.agent_id = producers.id
          AND p.seq > producers.after_seq
          AND p.context_item_id IS NOT NULL
          AND p.provenance_json IS NOT NULL
          AND p.public_import_exempt IS NOT TRUE
          AND p.public_import_done_provenance_md5
              IS DISTINCT FROM md5(p.provenance_json)::uuid
        OFFSET 0
    ) candidate
    UNION
    SELECT late.agent_id, late.seq, late.message_id
    FROM root_late_context late
),
-- Durable provenance is typed data. Surface corruption to Rust in this same
-- statement snapshot instead of replaying a consumed identity or silently
-- omitting history. The target/root NULL-key probes use migration 0399's index.
validation AS MATERIALIZED (
    SELECT CASE
      WHEN EXISTS (
        SELECT 1 FROM producer_context WHERE message_id IS NULL
      ) THEN 'producer'
      WHEN EXISTS (SELECT 1 FROM producer_context) AND EXISTS (
        SELECT 1
        FROM context_item_resume_projection seen
        WHERE seen.agent_id = $3
          AND seen.provenance_json IS NOT NULL
          AND md5(agent.resume_projection_message_id_v1(
              seen.provenance_json
          ))::uuid IS NULL
      ) THEN 'target'
      WHEN EXISTS (SELECT 1 FROM producer_context) AND EXISTS (
        SELECT 1
        FROM context_item_resume_projection seen
        WHERE $4::bigint IS NOT NULL
          AND seen.agent_id = $1
          AND seen.seq <= $4
          AND seen.provenance_json IS NOT NULL
          AND md5(agent.resume_projection_message_id_v1(
              seen.provenance_json
          ))::uuid IS NULL
      ) THEN 'root'
    END AS invalid_scope
),
unseen_context AS MATERIALIZED (
    SELECT candidate.agent_id, candidate.seq, candidate.message_id
    FROM producer_context candidate
    CROSS JOIN validation
    -- Keep separate per-agent digest probes. Combining the agent IDs with OR
    -- lets custom plans skip-scan the expression index by message ID.
    LEFT JOIN LATERAL (
        SELECT 1 AS hit
        FROM context_item_resume_projection seen
        WHERE seen.agent_id = $3
          AND seen.provenance_json IS NOT NULL
          -- A one-value range is an intentional custom-plan fence: equality
          -- can prefer the agent PK and scan every consumed target identity.
          AND md5(agent.resume_projection_message_id_v1(
              seen.provenance_json
          ))::uuid BETWEEN md5(candidate.message_id)::uuid
                         AND md5(candidate.message_id)::uuid
          AND agent.resume_projection_message_id_v1(
              seen.provenance_json
          ) = candidate.message_id
        LIMIT 1
    ) target_consumed ON true
    LEFT JOIN LATERAL (
        SELECT 1 AS hit
        FROM context_item_resume_projection seen
        WHERE $4::bigint IS NOT NULL
          AND seen.agent_id = $1
          AND seen.seq <= $4
          AND seen.provenance_json IS NOT NULL
          -- Match the target probe's expression-index plan fence.
          AND md5(agent.resume_projection_message_id_v1(
              seen.provenance_json
          ))::uuid BETWEEN md5(candidate.message_id)::uuid
                         AND md5(candidate.message_id)::uuid
          AND agent.resume_projection_message_id_v1(
              seen.provenance_json
          ) = candidate.message_id
        LIMIT 1
    ) root_consumed ON true
    WHERE validation.invalid_scope IS NULL
      AND candidate.message_id IS NOT NULL
      AND target_consumed.hit IS NULL
      AND root_consumed.hit IS NULL
),
feed AS MATERIALIZED (
SELECT context.agent_id AS producer_agent_id, context.seq AS producer_context_seq,
       ci.item_json, (extract(epoch FROM ci.created_at))::bigint AS created_at,
       projection.message_source, projection.client_context_json, projection.reply_prefix,
       projection.provenance_json, projection.client_context_projected,
       projection.reply_prefix_projected, projection.provenance_projected,
       e.event_seq AS sort_event_seq, ci.seq AS sort_context_seq
FROM unseen_context context
JOIN context_item_resume_projection projection
  ON projection.agent_id = context.agent_id AND projection.seq = context.seq
JOIN context_items ci
  ON ci.agent_id = context.agent_id AND ci.seq = context.seq
 AND ci.context_item_id = projection.context_item_id
-- Probe each unseen message by ID. OFFSET 0 keeps the planner from flattening
-- this into a scan of every event in the public conversation.
CROSS JOIN LATERAL (
    SELECT e.event_seq, e.event_name, e.payload_json
    FROM runtime.events e
    WHERE e.message_id = context.message_id AND e.parent_agent_id = $1
      AND COALESCE(NULLIF(TRIM(e.agent_id), ''), e.parent_agent_id) = context.agent_id
      AND e.event_seq <= $2
      AND e.transcript_surface = 'main_chat'
      AND ((e.visibility = 'public' AND (
            (e.event_name = 'message.user' AND e.role = 'user')
            OR (e.event_name = 'message.assistant' AND e.role = 'assistant')))
        OR (e.visibility = 'internal' AND e.event_name = 'message.internal'
            AND e.source = 'runtime.voice_companion_context'
            AND EXISTS (
              SELECT 1 FROM runtime.event_payload_fields role_field
              WHERE role_field.event_seq = e.event_seq
                AND role_field.field_path = '/role'
                AND role_field.scalar_type = 'string'
                AND COALESCE(
                  role_field.text_value, role_field.scalar_value
                ) = 'developer'
            )))
    OFFSET 0
) e
JOIN runtime.event_payload_fields context_seq
  ON context_seq.event_seq = e.event_seq AND context_seq.field_path = '/metadata/context_seq'
 AND COALESCE(context_seq.text_value, context_seq.scalar_value) = context.seq::text
LEFT JOIN runtime.event_payload_fields status
  ON status.event_seq = e.event_seq AND status.field_path = '/status'
LEFT JOIN runtime.event_payload_fields streaming
  ON streaming.event_seq = e.event_seq AND streaming.field_path = '/metadata/streaming'
LEFT JOIN runtime.event_payload_fields excluded
  ON excluded.event_seq = e.event_seq AND excluded.field_path = '/metadata/context_exclusion/reason'
WHERE (e.event_name <> 'message.assistant' OR COALESCE(
      status.text_value, status.scalar_value, e.payload_json::jsonb ->> 'status'
  ) IN ('completed', 'failed', 'interrupted'))
  AND streaming.event_seq IS NULL AND excluded.event_seq IS NULL
ORDER BY e.event_seq ASC, ci.seq ASC
LIMIT 256
)
SELECT feed.producer_agent_id, feed.producer_context_seq, feed.item_json,
       feed.created_at, feed.message_source, feed.client_context_json,
       feed.reply_prefix, feed.provenance_json, feed.client_context_projected,
       feed.reply_prefix_projected, feed.provenance_projected,
       feed.sort_event_seq, validation.invalid_scope
FROM validation
LEFT JOIN feed ON validation.invalid_scope IS NULL
WHERE validation.invalid_scope IS NOT NULL OR feed.producer_agent_id IS NOT NULL
ORDER BY feed.sort_event_seq ASC NULLS FIRST, feed.sort_context_seq ASC NULLS FIRST
