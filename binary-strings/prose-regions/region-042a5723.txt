       COALESCE(c.checkpoint_seq, -1) AS checkpoint_seq
FROM agent.agents AS a
LEFT JOIN LATERAL (
    SELECT checkpoint_seq
    FROM agent.agent_compactions
    WHERE agent_id = $1
      AND has_replacement_history
    ORDER BY id DESC
    LIMIT 1
) AS c ON c.checkpoint_seq > COALESCE(a.prompt_floor_seq, -1)
WHERE a.id = $1
