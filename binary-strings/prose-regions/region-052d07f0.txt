with carrier_fields as (
  select e.message_id,
         f.field_path,
         f.scalar_type,
         f.scalar_value,
         coalesce(f.text_value, f.scalar_value) as field_value
  from runtime.events e
  join runtime.event_payload_fields f on f.event_seq = e.event_seq
  where e.message_id = any($1)
    and e.event_name = 'message.user'
    and (
      f.field_path = '/reply_to_target/activation_mode'
      or f.field_path = '/reply_to_target/selected_item_ids'
      or f.field_path ~ '^/reply_to_target/selected_item_ids/[0-9]+$'
    )
)
select message_id,
       max(field_value) filter (
         where field_path = '/reply_to_target/activation_mode'
       ) as stored_activation_mode,
       coalesce(
         array_agg(field_value order by
           substring(field_path from '([0-9]+)$')::integer
         ) filter (
           where field_path ~ '^/reply_to_target/selected_item_ids/[0-9]+$'
             and nullif(btrim(field_value), '') is not null
         ),
         '{}'::text[]
       ) as selected_item_ids,
       coalesce(bool_or(
         field_path = '/reply_to_target/selected_item_ids'
         and scalar_type = 'array'
         and scalar_value = '[]'
       ), false) as explicit_empty_selection
from carrier_fields
group by message_id
