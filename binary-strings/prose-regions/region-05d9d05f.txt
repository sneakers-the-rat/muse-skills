FROM (
    SELECT
        action_index,
        action_id,
        activity_thread_id,
        agent_id,
        agent_depth,
        icon,
        title,
        subtitle,
        report,
        status,
        created_at,
        action_seq AS insertion_order
    FROM activity_monitor_thread_actions
    WHERE activity_thread_id = $1
    ORDER BY action_index DESC, insertion_order DESC
    LIMIT $2
)
ORDER BY action_index ASC, insertion_order ASC
