  var initialHydratedStateJson = __HATCH_INITIAL_WIDGET_STATE_JSON__;
  var selector = "input, textarea, select, [data-hatch-state-key][aria-pressed]";
  var formStateKey = "__hatch_form_state";
  var ignoredInputTypes = {
    button: true,
    file: true,
    hidden: true,
    image: true,
    password: true,
    reset: true,
    submit: true
  };
  var widgetRootSelector = "[data-widget-root], [data-share-widget-root], .hatch-widget";
  var currentScript = document.currentScript;

  function isWidgetRoot(el) {
    return el && el.matches && el.matches(widgetRootSelector);
  }

  function findWidgetRoot() {
    var candidate = currentScript ? currentScript.parentElement : null;
    while (candidate && candidate !== document.documentElement) {
      if (isWidgetRoot(candidate)) {
        return candidate;
      }
      candidate = candidate.parentElement;
    }

    candidate = currentScript ? currentScript.previousElementSibling : null;
    while (candidate) {
      if (isWidgetRoot(candidate)) {
        return candidate;
      }
      candidate = candidate.previousElementSibling;
    }

    return document.getElementById("root") || document;
  }

  var widgetRoot = findWidgetRoot();

  function rootElement() {
    return widgetRoot;
  }

  function containsTarget(el) {
    var root = rootElement();
    return root === document || root === el || (root.contains && root.contains(el));
  }

  function inputType(el) {
    return String(el.getAttribute("type") || el.type || "text").toLowerCase();
  }

  function isAriaPressedControl(el) {
    return (
      el &&
      el.hasAttribute &&
      el.hasAttribute("data-hatch-state-key") &&
      el.hasAttribute("aria-pressed")
    );
  }

  function isPersistable(el) {
    if (!el || !el.tagName) {
      return false;
    }
    if (isAriaPressedControl(el)) {
      return true;
    }
    var tag = String(el.tagName).toLowerCase();
    if (tag === "input") {
      return !ignoredInputTypes[inputType(el)];
    }
    return tag === "textarea" || tag === "select";
  }

  function controls() {
    return Array.prototype.slice
      .call(rootElement().querySelectorAll(selector))
      .filter(isPersistable);
  }

  function controlKey(el, index) {
    var explicit =
      el.getAttribute("data-hatch-state-key") ||
      el.getAttribute("data-state-key") ||
      el.id;
    if (explicit) {
      return explicit;
    }
    var tag = String(el.tagName || "").toLowerCase();
    if (tag === "input") {
      var type = inputType(el);
      var name = el.getAttribute("name") || "";
      var value = el.getAttribute("value") || "";
      if (type === "radio" && name) {
        return "radio:" + name + ":" + (value || index);
      }
      if (type === "checkbox" && name) {
        return "checkbox:" + name + ":" + (value || "on") + ":" + index;
      }
    }
    var fallbackName = el.getAttribute("name");
    return fallbackName || tag + ":" + index;
  }

  function normalizeState(value) {
    if (!value || typeof value !== "object" || Array.isArray(value)) {
      return {};
    }
    return value;
  }

  function copyState(value) {
    try {
      return normalizeState(JSON.parse(JSON.stringify(normalizeState(value))));
    } catch (_) {
      return {};
    }
  }

  function hasStateFields(value) {
    return Object.keys(normalizeState(value)).length > 0;
  }

  function currentBridgeState() {
    return window.hatchWidget && typeof window.hatchWidget.getState === "function"
      ? normalizeState(window.hatchWidget.getState())
      : {};
  }

  function initialHydratedState() {
    if (typeof initialHydratedStateJson !== "string") {
      return null;
    }
    try {
      return normalizeState(JSON.parse(initialHydratedStateJson));
    } catch (_) {
      return {};
    }
  }

  function valueFor(el) {
    if (isAriaPressedControl(el)) {
      return { pressed: el.getAttribute("aria-pressed") === "true" };
    }
    var tag = String(el.tagName || "").toLowerCase();
    if (tag === "input") {
      var type = inputType(el);
      if (type === "checkbox" || type === "radio") {
        return { checked: el.checked === true };
      }
    }
    if (tag === "select" && el.multiple) {
      return {
        value: Array.prototype.slice
          .call(el.options || [])
          .filter(function(option) {
            return option.selected;
          })
          .map(function(option) {
            return option.value;
          })
      };
    }
    return { value: String(el.value || "") };
  }

  function applyValue(el, saved) {
    if (!saved || typeof saved !== "object") {
      return;
    }
    if (isAriaPressedControl(el)) {
      el.setAttribute("aria-pressed", saved.pressed === true ? "true" : "false");
      return;
    }
    var tag = String(el.tagName || "").toLowerCase();
    if (tag === "input") {
      var type = inputType(el);
      if (type === "checkbox" || type === "radio") {
        el.checked = saved.checked === true;
        return;
      }
    }
    if (tag === "select" && el.multiple && Array.isArray(saved.value)) {
      var selected = {};
      saved.value.forEach(function(value) {
        selected[String(value)] = true;
      });
      Array.prototype.slice.call(el.options || []).forEach(function(option) {
        option.selected = selected[String(option.value)] === true;
      });
      return;
    }
    if (typeof saved.value === "string" && "value" in el) {
      el.value = saved.value;
    }
  }

  function formState(state) {
    var nested = normalizeState(state[formStateKey]);
    return Object.keys(nested).length > 0 || state[formStateKey]
      ? nested
      : state;
  }

  function restore(state) {
    var savedControls = formState(normalizeState(state));
    controls().forEach(function(el, index) {
      applyValue(el, savedControls[controlKey(el, index)]);
    });
  }

  function postState(state) {
    if (
      !window.hatchWidget ||
      typeof window.hatchWidget.setState !== "function"
    ) {
      return;
    }
    window.hatchWidget.setState(state);
  }

    var savedControls = {};
    controls().forEach(function(el, index) {
      savedControls[controlKey(el, index)] = valueFor(el);
    });
    var bridgeState = currentBridgeState();
    if (hasStateFields(bridgeState)) {
      lastKnownState = copyState(bridgeState);
    }
    var state = copyState(lastKnownState);
    state[formStateKey] = savedControls;
    lastKnownState = copyState(state);
    postState(state);
  }

    if (!target || !target.matches) {
      return null;

      return null;

      }
