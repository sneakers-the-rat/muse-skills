
select root_session_id, owner_agent_id, root_message_id, status, state_revision
from runtime.browser_tasks
where task_id = $1
  and admission_seq is not null
  and owner_kind is distinct from 'research'
select distinct node_id from runtime.browser_tasks where node_id is not null and admission_seq is not null and status in ('queued', 'starting', 'running', 'user_controlled', 'parked_outcome')current
