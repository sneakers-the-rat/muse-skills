WITH candidates AS MATERIALIZED (
  SELECT event.event_seq FROM runtime.events AS event
  WHERE event.display_text_ready = false
    AND event.event_name = 'message.assistant'
    AND event.visibility = 'public' AND event.role = 'assistant'
    AND ($1::text IS NULL OR event.message_id = $1)
    AND event.payload_json IS NOT NULL
    AND event.payload_json::jsonb ->> 'status' IN ('completed', 'failed', 'interrupted')
  ORDER BY event.event_seq LIMIT $2
), released AS (
  UPDATE runtime.events event SET display_text_ready = true
  FROM candidates WHERE event.event_seq = candidates.event_seq
  RETURNING event.event_seq
), publications AS (
  UPDATE chat_private.publications p SET body = m.body, display_ready = true,
    state = CASE WHEN t.disposition = 'quiet' THEN 'retired'
      WHEN t.disposition <> 'publish' THEN p.state
      WHEN length(trim(m.body)) = 0 THEN 'retired' ELSE 'pending' END
  FROM released JOIN runtime.messages m USING(event_seq), chat_private.publication_turns t
  WHERE p.event_seq = released.event_seq AND p.state = 'writing'
    AND t.execution_id = p.execution_id
  RETURNING p.event_seq
)
