    (() => {
      const root = document.currentScript.closest("[data-card-root]");
      if (!root) return;

      if (/Android/i.test(navigator.userAgent) && window.HatchBridge) {
        document.documentElement.setAttribute("data-nutrition-android-host", "true");
      }

      if (CSS.supports("-webkit-touch-callout", "none")) {
        const documentRoot = document.documentElement;
        const shellRoot = document.getElementById("root");
        const themeNodes = [documentRoot, document.body, shellRoot].filter(Boolean);
        const colorSchemeQuery = window.matchMedia("(prefers-color-scheme: dark)");
        const syncIosTheme = () => {
          const themes = themeNodes
            .map((node) => String(node.getAttribute("data-theme") || "").toLowerCase())
            .filter(Boolean);
          const explicitlyDark = themes.some((theme) => theme === "dark" || theme === "catppuccin-mocha");
          const explicitlyLight = themes.some((theme) => theme === "light");
          const classDark = themeNodes.some((node) => node.classList.contains("dark"));
          const shellColorScheme = shellRoot
            ? String(getComputedStyle(shellRoot).colorScheme || "").trim()
            : "";
          const dark = explicitlyDark
            || classDark
            || (!explicitlyLight && (shellColorScheme === "dark" || colorSchemeQuery.matches));
          documentRoot.setAttribute("data-nutrition-ios-theme", dark ? "dark" : "light");
        };

        syncIosTheme();
        const themeObserver = new MutationObserver(syncIosTheme);
        themeNodes.forEach((node) => {
          themeObserver.observe(node, {
            attributes: true,
            attributeFilter: ["class", "data-theme", "style"],
          });
        });
        colorSchemeQuery.addEventListener?.("change", syncIosTheme);
      }

      const dataEl = root.querySelector('script[type="application/json"][data-card-data]');
      let data = {};
      try { data = JSON.parse((dataEl && dataEl.textContent) || "{}"); } catch (_) {}
      root.lang = String(data.locale || "");
      root.dir = "auto";
      const expansion = root.querySelector("[data-ingredient-toggle]");
      if (expansion) expansion.textContent = String(data.view_more || "");
      const sectionLabel = root.querySelector("[data-section-label]");
      if (sectionLabel) sectionLabel.textContent = String(data.section_label || "");

      if (data.theme) root.setAttribute("data-accent", String(data.theme));

      const mealTitle = root.querySelector("[data-meal-title]");
      if (mealTitle) mealTitle.textContent = String(data.summary || "");

      const mealIcon = root.querySelector("[data-meal-icon]");
      const mealIconFrame = root.querySelector("[data-meal-icon-frame]");
      if (mealIcon && mealIconFrame && data.icon_data_url) {
        mealIcon.src = String(data.icon_data_url);
        mealIconFrame.hidden = false;
      }

      const surface = root.querySelector("[data-surface]");
      if (surface) {
        surface.setAttribute("role", "group");
        surface.setAttribute("aria-label", String(data.aria_label || ""));
      }

      const metricsMount = root.querySelector("[data-metrics]");
      if (metricsMount) {
        metricsMount.replaceChildren();
        const fragment = document.createDocumentFragment();
        (Array.isArray(data.metrics) ? data.metrics : []).slice(0, 4).forEach((metric) => {
          if (!metric || !metric.label) return;
          const item = document.createElement("div");
          item.className = "nutrition-metric";

          const label = document.createElement("div");
          label.className = "nutrition-metric-label";
          label.textContent = String(metric.label);

          const value = document.createElement("div");
          value.className = "nutrition-metric-value";
          value.textContent = String(metric.value || "-");

          item.append(label, value);
          fragment.append(item);
        });
        metricsMount.append(fragment);
      }

      const initialValue = Number(data.initial_visible_count);
      const initialVisibleCount = Number.isFinite(initialValue) && initialValue > 0
        ? Math.floor(initialValue)
        : 2;
      const ingredients = (Array.isArray(data.ingredients) ? data.ingredients : [])
        .filter((ingredient) => ingredient && ingredient.name);
      const rows = [];
      const ingredientsMount = root.querySelector("[data-ingredients]");

      if (ingredientsMount) {
        ingredientsMount.replaceChildren();
        const fragment = document.createDocumentFragment();
        ingredients.forEach((ingredient, index) => {
          const wrapper = document.createElement("div");
          wrapper.className = "ingredient-item";
          const collapsed = index >= initialVisibleCount;
          wrapper.classList.toggle("is-collapsed", collapsed);
          wrapper.setAttribute("aria-hidden", collapsed ? "true" : "false");

          const content = document.createElement("div");
          content.className = "ingredient-item-content";
          const row = document.createElement("div");
          row.className = "ingredient-row";

          const name = document.createElement("div");
          name.className = "ingredient-name";
          name.textContent = String(ingredient.name);
          row.append(name);
          if (ingredient.meta) {
            const meta = document.createElement("div");
            meta.className = "ingredient-meta";
            meta.textContent = String(ingredient.meta);
            row.append(meta);
          }

          content.append(row);
          wrapper.append(content);
          rows.push(wrapper);
          fragment.append(wrapper);
        });
        ingredientsMount.append(fragment);
      }

      const hasIngredients = rows.length > 0;
      const divider = root.querySelector("[data-divider]");
      const ingredientsSection = root.querySelector("[data-ingredients-section]");
      if (divider) divider.hidden = !hasIngredients;
      if (ingredientsSection) ingredientsSection.hidden = !hasIngredients;

      const toggle = root.querySelector("[data-ingredient-toggle]");
      if (toggle && rows.length > initialVisibleCount) {
        toggle.hidden = false;
        toggle.addEventListener("click", () => {
          rows.forEach((row) => {
            row.classList.remove("is-collapsed");
            row.setAttribute("aria-hidden", "false");
          });
          toggle.setAttribute("aria-expanded", "true");
          toggle.hidden = true;
        }, {once: true});
      }
    })();
