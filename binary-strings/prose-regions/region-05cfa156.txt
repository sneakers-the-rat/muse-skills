SELECT e.event_seq,
       NULLIF(BTRIM(COALESCE(
         tz.text_value,
         tz.scalar_value,
         e.payload_json::jsonb #>> '{metadata,client_timezone}'
       )), '') AS client_timezone
FROM runtime.events AS e
LEFT JOIN runtime.event_payload_fields AS tz ON tz.event_seq = e.event_seq
  AND tz.field_path = '/metadata/client_timezone'
WHERE e.event_name = 'message.user'
  AND e.visibility = 'public'
  AND e.source IN ('runtime', 'runtime.voice_audio')
  AND e.event_seq >= (
    SELECT min(seq)
    FROM (
      SELECT e2.event_seq AS seq
      FROM runtime.events AS e2
      WHERE e2.event_name = 'message.user'
        AND e2.visibility = 'public'
        AND e2.source IN ('runtime', 'runtime.voice_audio')
      ORDER BY e2.event_seq DESC
      LIMIT 100
    ) recent
  )
  AND NULLIF(BTRIM(COALESCE(
        tz.text_value,
        tz.scalar_value,
        e.payload_json::jsonb #>> '{metadata,client_timezone}'
      )), '') IS NOT NULL
ORDER BY e.event_seq DESC
LIMIT 1
