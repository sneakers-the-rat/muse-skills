           'checkpoint_custody', mailbox.checkpoint_custody,
           'submission_json', mailbox.submission_json
         )::text

         WHERE mailbox.agent_id = runtime_restart_checkpoints.agent_id
           AND mailbox.state IN ('accepted', 'attached')
         ORDER BY mailbox.accepted_ordinal, mailbox.submission_id
       ) AS mailbox_overlay_rows
FROM runtime_restart_checkpoints
WHERE agent_id = $1
