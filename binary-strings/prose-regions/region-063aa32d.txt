                    union all select child.invocation_id from runtime.browser_invocations child
                    join branch parent on child.parent_invocation_id = parent.invocation_id)
                    update runtime.browser_invocations set state = $3, suspended_state = null, result_json = $2,
                    terminal_reason = $4, finished_at = now(), updated_at = now(),
                    handoff_state = case when parent_invocation_id is null then $5::text end,
                    handoff_submission_id = case when parent_invocation_id is null then $6::text end
