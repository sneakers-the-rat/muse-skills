SELECT r.binding_id FROM agent.connection_resources r
                 JOIN agent.connections c USING(connection_id)
                 JOIN chat_private.peer_conversations p ON p.chat_id=c.connection_id::text
                 WHERE Àj AND r.host_muse=$1 AND r.resource_id=$2
                 ORDER BY r.binding_id LIMIT 1 FOR SHARE OF r,c,p
