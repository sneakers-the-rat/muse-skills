  metadata.chat_id AS session_id,
  metadata.pinned,
  metadata.pinned_order,
  roots.updated_at,
  roots.created_at
FROM chat.chats AS metadata
JOIN agents AS roots
  ON roots.session_id = metadata.chat_id
 AND roots.kind = 'root'
 AND roots.parent_agent_id IS NULL
 AND roots.ephemeral IS DISTINCT FROM TRUE
