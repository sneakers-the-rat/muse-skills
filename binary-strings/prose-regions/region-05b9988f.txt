UPDATE subagent_spawns AS current
SET completed_at = COALESCE(completed_at, $1),
    status = $2,
    deferred_terminal_status = NULL,
    -- Interrupted spawns carry no deliverable result, so settle them as seen.
    -- Without this, a now-finalized interrupted child re-pins its parent via the
    -- "completed-but-unseen" guard below, and the unseen-completion drain would
    -- emit a recovery follow-up for every superseded browser worker.
    -- reopen_terminal_subagent_spawn_rows_in_tx resets seen_at on resume.
    seen_at = CASE WHEN $2 = 'interrupted' THEN COALESCE(seen_at, $1) ELSE seen_at END
WHERE current.child_agent_id = $3
  AND current.completed_at IS NULL
  AND EXISTS (
    SELECT 1
    FROM agents AS c
    WHERE c.id = current.child_agent_id
      AND COALESCE(current.deferred_terminal_status, c.status)
          IN ('completed', 'errored', 'shutdown', 'interrupted')
  )
  -- A terminal error cannot incorporate a result that finishes later, so
  -- descendant freshness must not pin it after every descendant settles.
  AND ($2 = 'errored' OR COALESCE(
    (
      SELECT c.updated_at
      FROM agents AS c
      WHERE c.id = current.child_agent_id
    ),
    0
  ) >= COALESCE(
    (
      -- Interrupted children carry no deliverable result, so the parent need
      -- not have its agent row updated past their (synthetic) completion time.
      -- Without this exclusion, settling an interrupted child (completed_at =
      -- interruption time, typically just after an idle parent's last update)
      -- flips the parent from G1-pinned to pinned on this freshness guard, so
      -- it never finalizes and its monitor lane keeps the global sweep spinning.
      -- Mirrors the interrupted seen_at handling in the SET clause above.
      SELECT MAX(child.completed_at)
      FROM subagent_spawns AS child
      JOIN agents AS child_agent ON child_agent.id = child.child_agent_id
      WHERE child.parent_agent_id = current.child_agent_id
        AND COALESCE(child.deferred_terminal_status, child_agent.status)
            <> 'interrupted'
    ),
    0
  ))
  AND NOT EXISTS (
    SELECT 1
    FROM subagent_spawns AS child
    WHERE child.parent_agent_id = current.child_agent_id
      AND child.completed_at IS NULL
  )
  AND NOT EXISTS (
    SELECT 1
    FROM subagent_spawns AS child
    WHERE child.parent_agent_id = current.child_agent_id
      AND child.completed_at IS NOT NULL
      AND child.seen_at IS NULL
  )
RETURNING completed_at
