WITH boundary AS MATERIALIZED (
  SELECT GREATEST(
    COALESCE((SELECT prompt_floor_seq FROM agents WHERE id = $1), -1),
    COALESCE((SELECT checkpoint_seq FROM agent_compactions
              WHERE agent_id = $1 AND has_replacement_history
              ORDER BY id DESC LIMIT 1), -1)
  ) AS floor_seq
),
-- Provenance is a consumed-identity fence. Validate through migration 399's
-- expression index before deleting anything; treating malformed typed data as
-- absent would let a quarantined public voice item replay on a later import.
validation AS MATERIALIZED (
  SELECT EXISTS (
    SELECT 1
    FROM context_item_resume_projection invalid
    CROSS JOIN boundary
    WHERE invalid.agent_id = $1
      AND NOT (invalid.seq = ANY($2))
      AND invalid.seq > boundary.floor_seq
      AND invalid.provenance_json IS NOT NULL
      AND md5(agent.resume_projection_message_id_v1(
          invalid.provenance_json
      ))::uuid IS NULL
  ) AS invalid_provenance
  FROM boundary
),
-- Carry the statement-snapshot row version into DELETE. A concurrent event
-- writer changes ctid, so EvalPlanQual retains that projection rather
-- than deleting an identity whose newly committed event is not in our snapshot.
rewrite_scope AS MATERIALIZED (
  SELECT projection.ctid AS row_ctid,
         agent.resume_projection_message_id_v1(
           projection.provenance_json
         ) AS message_id
  FROM context_item_resume_projection projection
  CROSS JOIN boundary
  WHERE (SELECT NOT invalid_provenance FROM validation)
    AND projection.agent_id = $1
    AND NOT (projection.seq = ANY($2))
    AND projection.seq > boundary.floor_seq
),
event_scope AS NOT MATERIALIZED (
  SELECT candidate.row_ctid, candidate.message_id
  FROM rewrite_scope candidate
  WHERE candidate.message_id IS NOT NULL
),
scope_stats AS MATERIALIZED (
  SELECT count(*) FILTER (WHERE candidate.message_id IS NOT NULL) AS event_candidate_count,
         count(*) FILTER (WHERE candidate.message_id IS NULL) AS null_candidate_count
  FROM rewrite_scope candidate
),
-- Bound random event-index probes for small rewrites. For genuinely large
-- rewrites, leave the anti-join flattenable so PostgreSQL can hash events once.
deletable AS (
  SELECT branch.row_ctid
  FROM scope_stats scope
  CROSS JOIN LATERAL (
    SELECT candidate.row_ctid
    FROM rewrite_scope candidate
    WHERE scope.null_candidate_count > 0
      AND candidate.message_id IS NULL
    OFFSET 0
  ) branch
  UNION ALL
  SELECT branch.row_ctid
  FROM scope_stats scope
  CROSS JOIN LATERAL (
    SELECT candidate.row_ctid
    FROM event_scope candidate
    WHERE scope.event_candidate_count BETWEEN 1 AND 4096
      AND NOT EXISTS (
        SELECT 1 FROM runtime.events e
        WHERE e.message_id = candidate.message_id
          AND COALESCE(NULLIF(TRIM(e.agent_id), ''), e.parent_agent_id) <> $1
          AND e.transcript_surface = 'main_chat'
          AND (e.visibility = 'public'
            OR (e.visibility = 'internal'
              AND e.event_name = 'message.internal'
              AND e.source = 'runtime.voice_companion_context'
              AND NOT EXISTS (
                SELECT 1 FROM runtime.event_payload_fields role_field
                WHERE role_field.event_seq = e.event_seq
                  AND role_field.field_path = '/role'
                  AND role_field.scalar_type = 'string'
                  AND COALESCE(
                    role_field.text_value, role_field.scalar_value
                  ) <> 'developer'
              )))
        OFFSET 0
      )
    OFFSET 0
  ) branch
  UNION ALL
  SELECT branch.row_ctid
  FROM scope_stats scope
  CROSS JOIN LATERAL (
    SELECT candidate.row_ctid
    FROM event_scope candidate
    WHERE scope.event_candidate_count > 4096
      AND NOT EXISTS (
        SELECT 1 FROM runtime.events e
        WHERE e.message_id = candidate.message_id
          AND COALESCE(NULLIF(TRIM(e.agent_id), ''), e.parent_agent_id) <> $1
          AND e.transcript_surface = 'main_chat'
          AND (e.visibility = 'public'
            OR (e.visibility = 'internal'
              AND e.event_name = 'message.internal'
              AND e.source = 'runtime.voice_companion_context'
              AND NOT EXISTS (
                SELECT 1 FROM runtime.event_payload_fields role_field
                WHERE role_field.event_seq = e.event_seq
                  AND role_field.field_path = '/role'
                  AND role_field.scalar_type = 'string'
                  AND COALESCE(
                    role_field.text_value, role_field.scalar_value
                  ) <> 'developer'
              )))
      )
    OFFSET 0
  ) branch
),
deleted AS (
  DELETE FROM context_item_resume_projection projection
  WHERE projection.agent_id = $1
    AND projection.ctid = ANY(ARRAY(
      SELECT candidate.row_ctid
      FROM deletable candidate
    ))
)

          AND p.public_import_exempt IS NOT TRUE
          AND p.public_import_done_provenance_md5
              IS DISTINCT FROM md5(p.provenance_json)::uuid
        OFFSET 0
    ) pending
    CROSS JOIN LATERAL (
        SELECT a.id,
               CASE WHEN pg_input_is_valid(
                   a.spawn_metadata_json::jsonb #>> '{realtime_voice,root_seed_context_seq}',
                   'bigint'
               ) THEN GREATEST(
                   (a.spawn_metadata_json::jsonb #>> '{realtime_voice,root_seed_context_seq}')::bigint,
                   -1
               ) ELSE -1 END AS after_seq
        FROM agents a
        WHERE a.id = pending.agent_id
          AND a.presentation_root_session_id = public_root.id
          AND a.id <> public_root.id
          AND a.kind = 'root' AND a.depth = 0

          AND a.spawn_metadata_json::jsonb #> '{realtime_voice,prompt_floor_seq}'

    SELECT candidate.agent_id, candidate.seq, candidate.provenance_md5,
           candidate.message_id
    FROM producers

        SELECT p.agent_id, p.seq, md5(p.provenance_json)::uuid AS provenance_md5,
               agent.resume_projection_message_id_v1(
                   p.provenance_json

        FROM context_item_resume_projection p

          AND p.seq > producers.after_seq

-- Each stamp rewrites its row and every index entry, so one import stamps at
-- most 256 rows; a backlog stays pending (fully checked) for later imports.
held AS MATERIALIZED (
    SELECT pending.agent_id, pending.seq, pending.provenance_md5
    FROM pending
    -- LIMIT 1 keeps this a per-row index probe; as EXISTS, PostgreSQL can
    -- hash every root row through the message ID extractor instead.
    CROSS JOIN LATERAL (
        SELECT 1
        FROM context_item_resume_projection seen
        WHERE seen.agent_id = $1
          AND seen.provenance_json IS NOT NULL
          -- Same one-value range fence as the feed's consumed probes.
          AND md5(agent.resume_projection_message_id_v1(
              seen.provenance_json
          ))::uuid BETWEEN md5(pending.message_id)::uuid
                       AND md5(pending.message_id)::uuid
          AND agent.resume_projection_message_id_v1(
              seen.provenance_json
          ) = pending.message_id
        LIMIT 1
    ) root_holds
    WHERE pending.message_id IS NOT NULL
    LIMIT 256
)

invalid_scope-- Runs after a complete root import, in its transaction: a row of the root's
-- realtime voice agents that is still pending was neither imported nor already
-- held, so the feed's rules skip it. Once its agent has no live session ($2)
-- and the row has not changed for $3 seconds, nothing can make it importable:
-- unfinished, suppressed or excluded replies and other roles' messages. Leaving
-- it pending would keep every later import probe true. The batch bound keeps a
-- large backlog from holding the import past its statement timeout; later
-- imports take the rest.
WITH batch AS MATERIALIZED (
    -- Pending rows first, as in the import probe.
    SELECT stale.agent_id, stale.seq
    FROM (
        SELECT DISTINCT p.agent_id
        FROM context_item_resume_projection p
        WHERE p.provenance_json IS NOT NULL
          AND p.public_import_exempt IS NOT TRUE
          AND p.public_import_done_provenance_md5 IS DISTINCT FROM md5(p.provenance_json)::uuid
        OFFSET 0
    ) pending
    CROSS JOIN LATERAL (
        SELECT a.id
        FROM agents a
        WHERE a.id = pending.agent_id
          AND a.presentation_root_session_id = $1
          AND a.id <> $1
          AND a.kind = 'root' AND a.depth = 0 AND a.agent_type = 'realtime_voice'
          AND NOT (a.id = ANY($2::text[]))
        OFFSET 0
    ) producer
    CROSS JOIN LATERAL (
        SELECT p.agent_id, p.seq
        FROM context_item_resume_projection p
        WHERE p.agent_id = producer.id
          AND p.updated_at < now() - make_interval(secs => $3)
          AND p.provenance_json IS NOT NULL
          AND p.public_import_exempt IS NOT TRUE
          AND p.public_import_done_provenance_md5 IS DISTINCT FROM md5(p.provenance_json)::uuid
        OFFSET 0
    ) stale
    LIMIT 1000
)

SET public_import_exempt = true
FROM batch
WHERE p.agent_id = batch.agent_id AND p.seq = batch.seq
loaded public voice history-- Exempt one batch of rows written outside every import producer range: seed
-- copies at or below a call's bound, Companion and root history, and every
-- other agent's rows. They can never become import candidates (agent kind,
-- depth, type and spawn metadata never change), so the pending index stays
-- limited to call rows the root has not taken yet.
WITH pending_agents AS MATERIALIZED (
    SELECT DISTINCT p.agent_id
    FROM context_item_resume_projection p
    WHERE p.provenance_json IS NOT NULL
      AND p.public_import_exempt IS NOT TRUE
      AND p.public_import_done_provenance_md5
          IS DISTINCT FROM md5(p.provenance_json)::uuid
),
-- Parse spawn metadata once per agent. NULL: no row is in a producer range.
-- Like the feed, a call produces only while its prompt floor matches its
-- root's; floors only advance, so a call from before a clear never does again.
ranges AS MATERIALIZED (
    SELECT pending_agents.agent_id,
           CASE WHEN a.kind = 'root' AND a.depth = 0
                     AND a.agent_type = 'realtime_voice' THEN CASE
               WHEN NOT pg_input_is_valid(a.spawn_metadata_json, 'jsonb') THEN -1
               WHEN root.id IS NULL
                    OR a.spawn_metadata_json::jsonb #> '{realtime_voice,prompt_floor_seq}'
                       IS DISTINCT FROM to_jsonb(COALESCE(root.prompt_floor_seq, -1))
                   THEN NULL
               WHEN pg_input_is_valid(
                   a.spawn_metadata_json::jsonb #>> '{realtime_voice,root_seed_context_seq}',
                   'bigint'
               ) THEN GREATEST(
                   (a.spawn_metadata_json::jsonb #>> '{realtime_voice,root_seed_context_seq}')::bigint,
                   -1
               )
               ELSE -1
           END END AS producer_after_seq
    FROM pending_agents
    LEFT JOIN agents a ON a.id = pending_agents.agent_id
    LEFT JOIN agents root
      ON root.id = a.presentation_root_session_id AND root.id <> a.id
),
batch AS MATERIALIZED (
    SELECT candidate.agent_id, candidate.seq
    FROM ranges
    -- Probe only the rows below each producer range, never the call rows
    -- inside it that stay pending until the root takes them.
    CROSS JOIN LATERAL (
        SELECT p.agent_id, p.seq
        FROM context_item_resume_projection p
        WHERE p.agent_id = ranges.agent_id
          AND p.seq <= COALESCE(ranges.producer_after_seq, 9223372036854775807)
          AND p.provenance_json IS NOT NULL
          AND p.public_import_exempt IS NOT TRUE
          AND p.public_import_done_provenance_md5
              IS DISTINCT FROM md5(p.provenance_json)::uuid
        LIMIT $1
    ) candidate
    LIMIT $1
)
