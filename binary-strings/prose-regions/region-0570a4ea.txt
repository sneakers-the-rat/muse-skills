// Inert until a host enables pick mode. On a tap it describes the tapped element and
// hands the description to the host (iOS WKScriptMessageHandler "hatchSpacesElementPicker",
// Android WebMessageListener "HatchElementPicker", or, on web, the parent shell via
// postMessage "hatch:space:element-picked"). The host collects one note per pick and sends
// them all as the "artifact_annotations" context attachment on one chat message (see
// CONTRACT.md). The host keeps numbered marks on the picked elements with
// "setMarks([{pick_id, number, path, text, page_path}])"; path/text/page_path let a mark
// come back after the document reloads.
//
// Page data never leaves through postMessage except to HOST_ORIGIN, which only the web
// host fills in at injection time; native hosts leave the placeholder, which disables
// every postMessage path. State reaches same-origin child frames by direct calls; other
// frames are never told anything.
(function () {
  if (window.__hatchElementPicker) return;

  var HOST_ORIGIN = "__HATCH_ELEMENT_PICKER_HOST_ORIGIN__";
  var hostOrigin = null;
  try {
    var hostUrl = new URL(HOST_ORIGIN);
    if ((hostUrl.protocol === "https:" || hostUrl.protocol === "http:") && hostUrl.origin === HOST_ORIGIN) {
      hostOrigin = HOST_ORIGIN;
    }
  } catch (e) {}

  var SEMANTIC = {
    div: 1, section: 1, article: 1, main: 1, nav: 1, header: 1, footer: 1, aside: 1,
    p: 1, h1: 1, h2: 1, h3: 1, h4: 1, h5: 1, h6: 1, li: 1, ul: 1, ol: 1, button: 1, a: 1,
    img: 1, figure: 1, table: 1, tr: 1, td: 1, th: 1, form: 1, input: 1, textarea: 1,
    select: 1, label: 1, details: 1, summary: 1, video: 1, canvas: 1, svg: 1
  };
  var TAP_SLOP_PX = 12;
  var TAP_MAX_MS = 700;
  var RECENT_PICKS_MAX = 20;
  var MAX_PATH = 1200;
  var MAX_DESCRIPTOR = 120;
  var ELIDED = "\u2026";
  var COLOR = "#0a7cff";
  var frameToken = Math.random().toString(36).slice(2, 10);
  var pickCounter = 0;
  var enabled = false;
  var overlay = null;
  var inertStyle = null;
  var downPoint = null;
  var recentPicks = []; // [{pick_id, el}], newest last
  var marks = {}; // pick_id -> {el, number, outline, badge}
  var lastMarks = [];
  var repositionQueued = false;

  function tagOf(el) { return el && el.tagName ? String(el.tagName).toLowerCase() : ""; }

  // At most max UTF-16 units, never splitting a surrogate pair, with lone
  // surrogates dropped: JSON.stringify escapes those, and strict JSON parsers
  // (the daemon's among them) reject the whole message over one.
  function clean(s, max) {
    var out = "";
    for (var i = 0; i < s.length && out.length < max; i++) {
      var c = s.charCodeAt(i);
      if (c >= 0xD800 && c <= 0xDBFF) {
        var d = s.charCodeAt(i + 1);
        if (!(d >= 0xDC00 && d <= 0xDFFF)) continue;
        if (out.length + 2 > max) break;
        out += s.charAt(i) + s.charAt(i + 1);
        i++;
        continue;
      }
      if (c >= 0xDC00 && c <= 0xDFFF) continue;
      out += s.charAt(i);
    }
    return out;
  }

  function describe(el) {
    var tag = tagOf(el);
    var id = typeof el.id === "string" ? el.id.trim() : "";
    var classAttr = el.getAttribute ? el.getAttribute("class") : null;
    var cls = typeof classAttr === "string"
      ? classAttr.trim().split(/\s+/).filter(Boolean).slice(0, 3)
      : [];
    return clean(tag + (id ? "#" + id : "") + (cls.length ? "." + cls.join(".") : ""), MAX_DESCRIPTOR);
  }

  // A tap on a span deep inside a card should name the card-level element.
  function snap(el) {
    var c = el;
    while (c && c.nodeType === 1 && tagOf(c) !== "body" && tagOf(c) !== "html") {
      if (SEMANTIC[tagOf(c)]) return c;
      c = c.parentElement;
    }
    return el;
  }

  // At most MAX_PATH long. A deeper path keeps body and the steps nearest the element,
  // and replaces the outermost ancestors after body with one ELIDED step.
  function pathOf(el) {
    var parts = [];
    var c = el;
    while (c && c.nodeType === 1) {
      parts.unshift(describe(c));
      if (tagOf(c) === "body") break;
      c = c.parentElement;
    }
    var path = parts.join(" > ");
    if (path.length <= MAX_PATH) return path;
    var head = parts[0] + " > " + ELIDED + " > ";
    var tail = parts.slice(1);
    while (tail.length > 1 && head.length + tail.join(" > ").length > MAX_PATH) tail.shift();
    return head + tail.join(" > ");
  }

  function textOf(el) {
    var t = (el.innerText != null ? el.innerText : el.textContent) || "";
    return clean(String(t).replace(/\s+/g, " ").trim(), 200);
  }

  function round(v) { return Math.round(v); }

  function payloadFor(el, x, y) {
    var r = el.getBoundingClientRect();
    pickCounter += 1;
    var pickId = frameToken + "-" + pickCounter;
    remember(pickId, el);
    var data = {
      pick_id: pickId,
      element: describe(el),
      path: pathOf(el),
      viewport: { width: round(window.innerWidth), height: round(window.innerHeight) },
      point: { x: round(x), y: round(y) },
      rect: { x: round(r.left), y: round(r.top), width: round(r.width), height: round(r.height) }
    };
    var text = textOf(el);
    if (text) data.text = text;
    var pagePath = currentPagePath();
    if (pagePath != null) data.page_path = pagePath;
    return data;
  }

  function currentPagePath() {
    try { return location.pathname + location.hash; } catch (e) { return null; }
  }

  function pickedElement(pickId) {
    for (var i = recentPicks.length - 1; i >= 0; i--) {
      if (recentPicks[i].pick_id === pickId) return recentPicks[i].el;
    }
    return null;
  }

  function remember(pickId, el) {
    recentPicks.push({ pick_id: pickId, el: el });
    if (recentPicks.length > RECENT_PICKS_MAX) recentPicks.shift();
  }

  // Walks "path" (pathOf's output) down from body, following every child whose
  // descriptor matches the step, and returns the one element at the end whose text
  // equals "text" (or the only element, without text). Ambiguity returns null.
  function locate(path, text) {
    var body = document.body;
    var steps = String(path).split(" > ");
    if (!body || steps[0] !== describe(body)) return null;
    var frontier = [body];
    var i = 1;
    if (steps[1] === ELIDED) {
      if (steps.length < 3) return null;
      var all = body.getElementsByTagName("*");
      frontier = [];
      for (var a = 0; a < all.length && a < 20000 && frontier.length < 64; a++) {
        if (describe(all[a]) === steps[2]) frontier.push(all[a]);
      }
      i = 3;
    }
    for (; i < steps.length && frontier.length; i++) {
      var next = [];
      for (var j = 0; j < frontier.length && next.length < 64; j++) {
        var kids = frontier[j].children;
        for (var k = 0; k < kids.length; k++) {
          if (describe(kids[k]) === steps[i]) next.push(kids[k]);
        }
      }
      frontier = next;
    }
    var hits = typeof text === "string" && text
      ? frontier.filter(function (n) { return textOf(n) === text; })
      : frontier;
    return hits.length === 1 ? hits[0] : null;
  }

  function styleBox(s) {
    s.position = "fixed"; s.pointerEvents = "none"; s.boxSizing = "border-box";
    s.zIndex = "2147483647";
  }

  function ensureOverlay() {
    if (overlay && overlay.isConnected) return overlay;
    overlay = document.createElement("div");
    overlay.setAttribute("data-hatch-element-picker", "");
    overlay.setAttribute("aria-hidden", "true");
    var s = overlay.style;
    styleBox(s);
    s.border = "2px solid " + COLOR; s.borderRadius = "6px";
    s.background = "rgba(10,124,255,0.10)"; s.display = "none";
    (document.body || document.documentElement).appendChild(overlay);
    return overlay;
  }

  function placeAround(node, r) {
    node.style.left = (r.left - 3) + "px"; node.style.top = (r.top - 3) + "px";
    node.style.width = (r.width + 6) + "px"; node.style.height = (r.height + 6) + "px";
  }

  function highlight(el) {
    var o = ensureOverlay();
    placeAround(o, el.getBoundingClientRect());
    o.style.display = "block";
  }

  function hideOverlay() { if (overlay) overlay.style.display = "none"; }

  // Numbered marks on the elements the user has already left notes on.
  function createMark(el, number) {
    var root = document.body || document.documentElement;
    var outline = document.createElement("div");
    outline.setAttribute("data-hatch-element-picker", "");
    outline.setAttribute("aria-hidden", "true");
    styleBox(outline.style);
    outline.style.border = "2px solid " + COLOR; outline.style.borderRadius = "6px";
    var badge = document.createElement("div");
    badge.setAttribute("data-hatch-element-picker", "");
    badge.setAttribute("aria-hidden", "true");
    var b = badge.style;
    styleBox(b);
    b.width = "22px"; b.height = "22px"; b.borderRadius = "11px"; b.background = COLOR;
    b.color = "#fff"; b.font = "700 12px/22px -apple-system, system-ui, sans-serif";
    b.textAlign = "center"; b.boxShadow = "0 1px 3px rgba(0,0,0,0.35)";
    badge.textContent = String(number);
    root.appendChild(outline); root.appendChild(badge);
    return { el: el, number: number, outline: outline, badge: badge };
  }

  function positionMark(m) {
    if (!m.el.isConnected) { m.outline.style.display = "none"; m.badge.style.display = "none"; return; }
    var r = m.el.getBoundingClientRect();
    m.outline.style.display = "block"; m.badge.style.display = "block";
    placeAround(m.outline, r);
    m.badge.style.left = (r.left - 11) + "px"; m.badge.style.top = (r.top - 11) + "px";
  }

  function positionAllMarks() {
    repositionQueued = false;
    for (var id in marks) if (Object.prototype.hasOwnProperty.call(marks, id)) positionMark(marks[id]);
  }

  function queueReposition() {
    if (repositionQueued) return;
    repositionQueued = true;
    (window.requestAnimationFrame || window.setTimeout)(positionAllMarks);
  }

  function removeMark(id) {
    var m = marks[id];
    if (!m) return;
    if (m.outline.parentNode) m.outline.parentNode.removeChild(m.outline);
    if (m.badge.parentNode) m.badge.parentNode.removeChild(m.badge);
    delete marks[id];
  }

  // Declarative: "list" is the complete set of marks the host wants shown. A pick id
  // belongs to one document; after a reload or an in-page navigation the host still
  // holds its notes, so an item may also carry the pick's "path", "text" and
  // "page_path", and an unknown id is re-resolved when exactly one element on the same
  // page matches. Ids that resolve to nothing here (another frame's or another page's
  // picks) are ignored.
  function setMarks(list) {
    list = Array.isArray(list) ? list : [];
    lastMarks = list;
    var keep = {};
    for (var i = 0; i < list.length; i++) {
      var item = list[i] || {};
      var id = typeof item.pick_id === "string" ? item.pick_id : null;
      var number = typeof item.number === "number" ? item.number : null;
      if (!id || number == null) continue;
      var el = pickedElement(id);
      if (!el && marks[id] && marks[id].el.isConnected) el = marks[id].el;
      if (!el && typeof item.path === "string" &&
          (typeof item.page_path !== "string" || item.page_path === currentPagePath())) {
        el = locate(item.path, item.text);
        if (el) remember(id, el);
      }
      if (!el) continue;
      keep[id] = true;
      if (marks[id] && marks[id].number !== number) {
        marks[id].number = number; marks[id].badge.textContent = String(number);
      } else if (!marks[id]) {
        marks[id] = createMark(el, number);
      }
    }
    for (var existing in marks) {
      if (Object.prototype.hasOwnProperty.call(marks, existing) && !keep[existing]) removeMark(existing);
    }
    positionAllMarks();
    hideOverlay();
    relay("setMarks", list);
  }

  // Same-origin child frames only: reading a property of a cross-origin window throws, so
  // those frames are skipped and learn nothing. (about:blank and srcdoc frames report a
  // "null" location origin while being same-origin, so this access check is the test.)
  function relay(method, arg) {
    for (var i = 0; i < window.frames.length; i++) {
      try {
        var api = window.frames[i].__hatchElementPicker;
        if (api && typeof api[method] === "function") api[method](arg);
      } catch (e) {}
    }
  }

  function deliver(data) {
    try {
      var wk = window.webkit && window.webkit.messageHandlers && window.webkit.messageHandlers.hatchSpacesElementPicker;
      if (wk) { wk.postMessage(data); return; }
    } catch (e) {}
    try {
      if (window.HatchElementPicker && typeof window.HatchElementPicker.postMessage === "function") {
        window.HatchElementPicker.postMessage(JSON.stringify(data)); return;
      }
    } catch (e) {}
    // Only the frame whose parent is the shell can match hostOrigin; postMessage drops the
    // message for any other parent.
    if (hostOrigin && window.parent && window.parent !== window) {
      window.parent.postMessage({ type: "hatch:space:element-picked", payload: data }, hostOrigin);
    }
  }

  function isOurs(node) {
    return !!(node && node.getAttribute && node.hasAttribute("data-hatch-element-picker"));
  }

  function targetAt(x, y) {
    var t = document.elementFromPoint(x, y);
    if (!t || isOurs(t)) return null;
    return snap(t);
  }

  // In pick mode the page is inert: no listener of the page sees a press, a click or a
  // menu gesture, and nothing activates, focuses or selects. Scrolling still works, since
  // nothing cancels touchstart or the moves; a press that moves more than TAP_SLOP_PX or
  // lasts longer than TAP_MAX_MS is a scroll or a long-press, not a pick.
  function swallow(ev) {
    if (!enabled) return;
    ev.stopImmediatePropagation();
    if (ev.cancelable && ev.type !== "touchstart") ev.preventDefault();
  }

  function onPointerDown(ev) {
    if (!enabled) return;
    ev.stopImmediatePropagation();
    if (ev.isPrimary === false) return;
    downPoint = { x: ev.clientX, y: ev.clientY, t: Date.now() };
    if (ev.pointerType === "mouse") { var t = targetAt(ev.clientX, ev.clientY); if (t) highlight(t); }
  }

  function onPointerMove(ev) {
    if (!enabled || ev.pointerType !== "mouse") return;
    var t = targetAt(ev.clientX, ev.clientY);
    if (t) highlight(t);
  }

  function onPointerUp(ev) {
    if (!enabled) return;
    swallow(ev);
    var start = downPoint; downPoint = null;
    if (!start || ev.isPrimary === false) return;
    if (Math.abs(ev.clientX - start.x) > TAP_SLOP_PX || Math.abs(ev.clientY - start.y) > TAP_SLOP_PX) return;
    if (Date.now() - start.t > TAP_MAX_MS) return;
    var el = targetAt(ev.clientX, ev.clientY);
    if (!el) return;
    highlight(el);
    deliver(payloadFor(el, ev.clientX, ev.clientY));
  }

  function onPointerCancel(ev) {
    if (!enabled) return;
    ev.stopImmediatePropagation();
    downPoint = null;
  }

  // Registered once at document start, before any page script, so these window capture
  // listeners always run ahead of the page's own; each one checks "enabled".
  var active = { capture: true, passive: false };
  var passive = { capture: true, passive: true };
  var SWALLOWED = ["mousedown", "mouseup", "click", "dblclick", "auxclick", "contextmenu", "dragstart", "touchend"];
  window.addEventListener("pointerdown", onPointerDown, active);
  window.addEventListener("pointermove", onPointerMove, passive);
  window.addEventListener("pointerup", onPointerUp, active);
  window.addEventListener("pointercancel", onPointerCancel, active);
  window.addEventListener("touchstart", swallow, passive);
  for (var si = 0; si < SWALLOWED.length; si++) window.addEventListener(SWALLOWED[si], swallow, active);

  function setInert(on) {
    if (on) {
      inertStyle = document.createElement("style");
      inertStyle.setAttribute("data-hatch-element-picker", "");
      inertStyle.textContent = "*{-webkit-user-select:none!important;user-select:none!important;-webkit-touch-callout:none!important}";
      (document.head || document.documentElement).appendChild(inertStyle);
    } else if (inertStyle) {
      if (inertStyle.parentNode) inertStyle.parentNode.removeChild(inertStyle);
      inertStyle = null;
    }
  }

  function setEnabled(on) {
    on = !!on;
    if (on !== enabled) {
      enabled = on;
      setInert(on);
      if (!on) {
        downPoint = null;
        hideOverlay();
      }
    }
    relay("setEnabled", on);
  }

  document.addEventListener("scroll", queueReposition, passive);
  window.addEventListener("resize", queueReposition);

  // Web: the shell toggles pick mode and sets the marks from hostOrigin. Native hosts call
  // the API directly in the main frame.
  window.addEventListener("message", function (ev) {
    var d = ev.data;
    if (!hostOrigin || !d || d.type !== "hatch:space:element-picker") return;
    if (ev.source !== window.parent || ev.origin !== hostOrigin) return;
    if (typeof d.enabled === "boolean") setEnabled(d.enabled);
    if (Array.isArray(d.marks)) setMarks(d.marks);
  });

  var api = Object.freeze({
    setEnabled: setEnabled,
    isEnabled: function () { return enabled; },
    setMarks: setMarks,
    currentMarks: function () { return lastMarks.slice(); },
    clearHighlight: function () { hideOverlay(); relay("clearHighlight"); }
  });
  Object.defineProperty(window, "__hatchElementPicker", {
    value: api, writable: false, configurable: false, enumerable: false
  });

  // A same-origin child frame that loads after the host spoke adopts its parent's state
  // once its own document exists.
  function adoptParentState() {
    try {
      var parentApi = window.parent !== window ? window.parent.__hatchElementPicker : null;
      if (!parentApi || parentApi === api) return;
      if (parentApi.isEnabled()) setEnabled(true);
      setMarks(parentApi.currentMarks());
    } catch (e) {}
  }
  if (window.parent !== window) {
    if (document.readyState === "loading") {
      document.addEventListener("DOMContentLoaded", adoptParentState, { once: true });
    } else {
      adoptParentState();
    }
  }
})();
