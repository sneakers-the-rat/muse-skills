        const colorSchemeQuery = window.matchMedia("(prefers-color-scheme: dark)");
        const syncIosTheme = () => {
          const themes = themeNodes
            .map((node) => String(node.getAttribute("data-theme") || "").toLowerCase())
            .filter(Boolean);
          const explicitlyDark = themes.some((theme) => theme === "dark" || theme === "catppuccin-mocha");
          const explicitlyLight = themes.some((theme) => theme === "light");
          const classDark = themeNodes.some((node) => node.classList.contains("dark"));
          const shellColorScheme = shellRoot
            ? String(getComputedStyle(shellRoot).colorScheme || "").trim()
            : "";
          const dark = explicitlyDark

  <script>
    (() => {
      const root = document.currentScript.closest("[data-card-root]");
      if (!root) return;

      if (/Android/i.test(navigator.userAgent) && window.HatchBridge) {
        document.documentElement.setAttribute("data-nutrition-android-host", "true");
      }

      if (CSS.supports("-webkit-touch-callout", "none")) {
        const documentRoot = document.documentElement;
        const shellRoot = document.getElementById("root");
        const themeNodes = [documentRoot, document.body, shellRoot].filter(Boolean);
        const colorSchemeQuery = window.matchMedia("(prefers-color-scheme: dark)");
        const syncIosTheme = () => {
          const themes = themeNodes
            .map((node) => String(node.getAttribute("data-theme") || "").toLowerCase())
            .filter(Boolean);
          const explicitlyDark = themes.some((theme) => theme === "dark" || theme === "catppuccin-mocha");
          const explicitlyLight = themes.some((theme) => theme === "light");
          const classDark = themeNodes.some((node) => node.classList.contains("dark"));
          const shellColorScheme = shellRoot
            ? String(getComputedStyle(shellRoot).colorScheme || "").trim()
            : "";
          const dark = explicitlyDark
            || classDark
            || (!explicitlyLight && (shellColorScheme === "dark" || colorSchemeQuery.matches));
          documentRoot.setAttribute("data-nutrition-ios-theme", dark ? "dark" : "light");
        };

        syncIosTheme();
        const themeObserver = new MutationObserver(syncIosTheme);
        themeNodes.forEach((node) => {
          themeObserver.observe(node, {
            attributes: true,
            attributeFilter: ["class", "data-theme", "style"],
          });
        });
        colorSchemeQuery.addEventListener?.("change", syncIosTheme);
      }

      const dataEl = root.querySelector('script[type="application/json"][data-card-data]');
      let data = {};
      try { data = JSON.parse((dataEl && dataEl.textContent) || "{}"); } catch (_) {}
      root.lang = String(data.locale || "");
      root.dir = "auto";
      const expansion = root.querySelector("[data-collection-toggle]");
      if (expansion) expansion.textContent = String(data.view_more || "");

      if (data.theme === "discovery") {
        root.setAttribute("data-accent", "discovery");
      }

      const sectionLabel = root.querySelector("[data-section-label]");
      if (sectionLabel) {
        sectionLabel.textContent = String(data.section_label || "");
      }

      const surface = root.querySelector("[data-surface]");
      if (surface) {
        surface.setAttribute("role", "group");
        surface.setAttribute("aria-label", String(data.aria_label || ""));
      }

      const metricsMount = root.querySelector("[data-metrics]");
      if (metricsMount) {
        metricsMount.replaceChildren();
        const fragment = document.createDocumentFragment();
        (Array.isArray(data.metrics) ? data.metrics : []).slice(0, 4).forEach((metric) => {
          if (!metric || !metric.label) return;
          const item = document.createElement("div");
          item.className = "nutrition-metric";

          const label = document.createElement("div");
          label.className = "nutrition-metric-label";
          label.textContent = String(metric.label);

          const value = document.createElement("div");
          value.className = "nutrition-metric-value";
          value.textContent = String(metric.value || "-");

          item.append(label, value);
          fragment.append(item);
        });
        metricsMount.append(fragment);
      }

      const renderMeal = (item) => {
        const wrapper = document.createElement("div");
        wrapper.className = "meal-history-item";

        const content = document.createElement("div");
        content.className = "meal-history-item-content";

        const row = document.createElement("div");
        row.className = "meal-history-row";

        const body = document.createElement("div");
        body.className = "meal-history-body";

        const title = document.createElement("div");
        title.className = "meal-history-title";
        title.textContent = String(item.summary || "");
        body.append(title);

        if (item.meta) {
          const meta = document.createElement("div");
          meta.className = "meal-history-meta";
          meta.textContent = String(item.meta);
          body.append(meta);
        }
        row.append(body);

        if (item.photo_data_url) {
          const photo = document.createElement("img");
          photo.className = "meal-history-photo";
          photo.alt = "";
          photo.addEventListener("error", () => {
            photo.remove();
          }, {once: true});
          photo.src = String(item.photo_data_url);
          row.append(photo);
        }

        content.append(row);
        wrapper.append(content);
        return wrapper;
      };

      const initialValue = Number(data.initial_visible_count);
      const initialVisibleCount = Number.isFinite(initialValue) && initialValue > 0
        ? Math.floor(initialValue)
        : 3;
      const allRows = [];
      const sections = [];
      const daysMount = root.querySelector("[data-days]");

      if (daysMount) {
        daysMount.replaceChildren();
        const fragment = document.createDocumentFragment();
        let renderedCount = 0;

        (Array.isArray(data.days) ? data.days : []).forEach((day) => {
          if (!day) return;
          const meals = (Array.isArray(day.meals) ? day.meals : [])
            .filter((meal) => meal && meal.summary);
          if (!meals.length) return;

          const section = document.createElement("section");
          section.className = "meal-day";

          const sectionContent = document.createElement("div");
          sectionContent.className = "meal-day-content";

          if (day.title) {
            const heading = document.createElement("h3");
            heading.className = "meal-day-title";
            heading.textContent = String(day.title);
            sectionContent.append(heading);
          }

          const list = document.createElement("div");
          list.className = "meal-day-list";
          const sectionRows = [];
          meals.forEach((meal) => {
            const row = renderMeal(meal);
            const collapsed = renderedCount >= initialVisibleCount;
            row.classList.toggle("is-collapsed", collapsed);
            row.setAttribute("aria-hidden", collapsed ? "true" : "false");
            renderedCount += 1;
            sectionRows.push(row);
            allRows.push(row);
            list.append(row);
          });
          sectionContent.append(list);
          section.append(sectionContent);
          sections.push({section, rows: sectionRows});
          fragment.append(section);
        });

        daysMount.append(fragment);
        daysMount.hidden = allRows.length === 0;
      }

      const syncSections = () => {
        sections.forEach(({section, rows}) => {
          const collapsed = !rows.some((row) => !row.classList.contains("is-collapsed"));
          section.classList.toggle("is-collapsed", collapsed);
          section.setAttribute("aria-hidden", collapsed ? "true" : "false");
        });
      };
      syncSections();

      const toggle = root.querySelector("[data-collection-toggle]");
      if (toggle && allRows.length > initialVisibleCount) {
        toggle.hidden = false;
        toggle.addEventListener("click", () => {
          allRows.forEach((row) => {
            row.classList.remove("is-collapsed");
            row.setAttribute("aria-hidden", "false");
          });
          syncSections();
          toggle.setAttribute("aria-expanded", "true");
          toggle.hidden = true;
        }, {once: true});
      }

      const emptyMount = root.querySelector("[data-empty]");
      if (emptyMount) {
        emptyMount.hidden = allRows.length > 0;
        emptyMount.textContent = data.empty_text || "";
      }
    })();
  </script>
