        const themeNodes = [documentRoot, document.body, shellRoot].filter(Boolean);
        const colorSchemeQuery = window.matchMedia("(prefers-color-scheme: dark)");
        const syncIosTheme = () => {
          const themes = themeNodes
            .map((node) => String(node.getAttribute("data-theme") || "").toLowerCase())
            .filter(Boolean);
          const explicitlyDark = themes.some((theme) => theme === "dark" || theme === "catppuccin-mocha");
          const explicitlyLight = themes.some((theme) => theme === "light");
          const classDark = themeNodes.some((node) => node.classList.contains("dark"));
          const shellColorScheme = shellRoot
            ? String(getComputedStyle(shellRoot).colorScheme || "").trim()
            : "";
          const dark = explicitlyDark
            || classDark
            || (!explicitlyLight && (shellColorScheme === "dark" || colorSchemeQuery.matches));
          documentRoot.setAttribute("data-nutrition-ios-theme", dark ? "dark" : "light");
        };

        syncIosTheme();
        const themeObserver = new MutationObserver(syncIosTheme);
        themeNodes.forEach((node) => {
          themeObserver.observe(node, {
            attributes: true,
            attributeFilter: ["class", "data-theme", "style"],
          });
        });
        colorSchemeQuery.addEventListener?.("change", syncIosTheme);
      }

      const dataEl = root.querySelector('script[type="application/json"][data-card-data]');

      root.lang = String(data.locale || "");

      if (expansion) expansion.textContent = String(data.view_more || "");
      const sectionLabel = root.querySelector("[data-section-label]");
      if (sectionLabel) sectionLabel.textContent = String(data.section_label || "");

      if (data.theme) root.setAttribute("data-accent", String(data.theme));

      const mealTitle = root.querySelector("[data-meal-title]");
      if (mealTitle) mealTitle.textContent = String(data.summary || "");

      const mealIcon = root.querySelector("[data-meal-icon]");
      const mealIconFrame = root.querySelector("[data-meal-icon-frame]");
      if (mealIcon && mealIconFrame && data.icon_data_url) {

            meta.textContent = String(ingredient.meta);
            row.append(meta);

          detail: { state: copyState(hydratedState) }
        })
      );
    } finally {
      seeding = false;
    }
  }

  function onWidgetState(event) {
    if (seeding) {
      return;
    }
    restoreGetState();
    window.removeEventListener("hatch-widget-state", onWidgetState);

    var incomingState = normalizeState(
      event && event.detail && event.detail.state
    );
    if (
      Object.keys(incomingState).length === 0 &&
      Object.keys(hydratedState).length > 0
    ) {
      if (event && typeof event.stopImmediatePropagation === "function") {
        event.stopImmediatePropagation();
      }
      dispatchHydratedState();
    }
    delete window.__hatchWidgetStateHydration;
  }

  window.addEventListener("hatch-widget-state", onWidgetState);
  window.__hatchWidgetStateHydration = {
    dispatch: dispatchHydratedState,
    destroy: function() {
      restoreGetState();
      window.removeEventListener("hatch-widget-state", onWidgetState);
    }
  };
})();(function() {
  var hydration = window.__hatchWidgetStateHydration;
  if (hydration && typeof hydration.dispatch === "function") {
    hydration.dispatch();
  }
})();
