SELECT agents.session_id, checkpoint.execution_config_json
FROM agent.chat_preferences preferences
JOIN agent.agents agents ON agents.id = preferences.active_root_id
JOIN agent.runtime_restart_checkpoints checkpoint ON checkpoint.agent_id = agents.id
WHERE preferences.id = 1
  AND agents.id = $1
  AND agents.agent_id = $1
  AND agents.kind = 'root'
  AND agents.parent_id IS NULL
  AND agents.parent_agent_id IS NULL
  AND agents.depth = 0
  AND agents.agent_type IS NULL
  AND agents.status = 'completed'
  AND agents.session_id IS NOT NULL
  AND checkpoint.mode = 'idle'
  AND checkpoint.payload_json IS NULL
  AND checkpoint.recovery_class = 'foreground_root'
  AND checkpoint.recovery_owner_id = $1
  AND checkpoint.execution_config_json IS NOT NULL
  AND NOT EXISTS (
    SELECT 1 FROM agent.message_mailbox
    WHERE agent_id = $1 AND state IN ('accepted', 'attached')
  )
  AND NOT EXISTS (
    SELECT 1 FROM agent.mail_conversations WHERE agent_id = $1
  )
  AND NOT EXISTS (
    SELECT 1 FROM agent.chat_workers WHERE agent_id = $1
  )
  AND NOT EXISTS (
    SELECT 1 FROM agent.delegate_requests WHERE worker_agent_id = $1
  )
  AND NOT EXISTS (
    SELECT 1 FROM agent.recovery_owners
    WHERE agent_id = $1 AND terminal_at IS NULL
  )
  AND NOT EXISTS (
    SELECT 1 FROM runtime.workflow_agent_calls calls
    JOIN runtime.workflow_runs runs ON runs.run_id = calls.run_id
    LEFT JOIN runtime.workflow_phase_runs phases
      ON phases.phase_run_id = calls.phase_run_id
    WHERE calls.child_agent_id = $1
      AND calls.status IN ('pending', 'running')
      AND runs.status IN ('queued', 'running', 'paused')
      AND (calls.phase_run_id IS NULL OR phases.status IN ('pending', 'running', 'paused'))
  )
  AND NOT EXISTS (
    SELECT 1 FROM self_improvement.run_steps
    WHERE agent_id = $1 AND status = 'running'
  )
  AND NOT EXISTS (
    SELECT 1 FROM background_workers.turns
    WHERE agent_id = $1 AND state = 'running'
  )
  AND NOT EXISTS (
    SELECT 1 FROM feed.run_steps
    WHERE agent_id = $1 AND status = 'running'
  )
  AND NOT EXISTS (
    SELECT 1 FROM scheduler.job_runs
    WHERE worker_agent_id = $1 AND status IN ('queued', 'running')
  )
  AND NOT EXISTS (
    SELECT 1 FROM runtime.execute_resolve_runs
    WHERE worker_agent_id = $1 AND status IN ('pending', 'notify_pending')
  )
