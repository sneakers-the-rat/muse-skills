), maintenance_obligations as (
  -- Count queued and running debt, including deadlines too near to register.
  -- RuntimeWork liveness is sampled separately: a row crossing its deadline
  -- or being claimed between those reads must remain visible in this snapshot.
  -- Read the earliest deadline so a successor seeded after projection
  -- reconciliation cannot be covered by an older, later alarm.
  select min(deadline_at) as next_deadline_at
  from runtime.work_items
  where work_class = 'maintenance_job'
    and terminalized_at is null
    and state in ('queued', 'running')
    -- A paused maintenance budget runs no queued row, so it owes no wake.
    and not ($5::boolean and state = 'queued')
    and deadline_at is not null
    -- Mirrors the wake predicate in `scheduled_resume.rs`.
    and coalesce(
          (metadata_json::jsonb)->>'suspend_policy' = 'wake_to_complete',
          ((metadata_json::jsonb)->>'wake_eligible')::boolean
        )
), scheduled_resume as (
  -- The VM holds a wake for the current projection only when a registration
  -- covers that exact instant and no live maintenance debt precedes it.
  -- Registration is additive, so other outstanding registrations for later
  -- instants are deliberately not reported here: a 7pm registration says
  -- nothing about a 3pm obligation. Shadow registrations count once the
  -- endpoint accepts and journals them, even though they do not arm a timer.
  select s.projection_complete,
         case
           when s.projection_complete
             and (maintenance_obligations.next_deadline_at is null
                  or maintenance_obligations.next_deadline_at
                     >= to_timestamp(s.next_resume_at_utc::double precision))

                   from scheduler.scheduled_resume_registrations r
                  where r.resume_at_utc = s.next_resume_at_utc
                    and r.schedule_id is not null)
           else null
         end as registered_dispatch_at_ms
  from scheduler.scheduled_resume_state s
  cross join maintenance_obligations
