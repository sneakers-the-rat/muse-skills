  UPDATE subagent_spawns
SET completed_at = COALESCE(completed_at, $1),
    status = $2,
    deferred_terminal_status = NULL,
    final_response = COALESCE(final_response, $3),
    seen_at = CASE WHEN $2 = 'interrupted' THEN COALESCE(seen_at, $1) ELSE seen_at END
WHERE child_agent_id = $4
    AND EXISTS (SELECT 1 FROM ownership_lock)
    AND (
      NOT $6
      OR (completed_at IS NULL AND deferred_terminal_status IS NULL)
    )
    AND EXISTS (
      SELECT 1 FROM agents
      WHERE agents.id = subagent_spawns.child_agent_id
        AND (
          ($6 AND agents.status IN ('pending_init', 'running'))
          OR (NOT $6 AND agents.status NOT IN ('shutdown', 'not_found'))
        )
    )
  RETURNING child_agent_id
)
