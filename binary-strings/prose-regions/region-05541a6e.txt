WITH candidates AS (
  SELECT sp.child_agent_id,
         sp.requester_source,
         COALESCE(child.status, sp.status) AS child_status,
         parent.status AS parent_status,
         sp.completed_at,
         CASE
           WHEN sp.requester_source = 'scheduler.cron'
             THEN 'scheduler_cron_stale_root'
           ELSE 'old_terminal_non_cron_root'
         END AS prune_reason
  FROM subagent_spawns AS sp
  JOIN agents AS parent
    ON parent.id = sp.parent_agent_id
  LEFT JOIN agents AS child
    ON child.id = sp.child_agent_id
  WHERE sp.completed_at IS NOT NULL
    AND sp.seen_at IS NULL
    AND parent.kind = 'root'
    AND (
      (
        sp.requester_source = 'scheduler.cron'
        AND sp.created_at <= $1
        AND sp.completed_at <= $1
        AND (
          parent.updated_at < sp.completed_at
          OR parent.status IN ('completed', 'errored', 'shutdown', 'not_found')
        )
      )
      OR (
        sp.requester_source IN ('runtime', 'runtime.browser_monitor')
        AND sp.created_at <= $2
        AND sp.completed_at <= $2
        AND COALESCE(child.status, sp.status) IN ('shutdown', 'interrupted', 'errored')
      )
    )
    AND NOT EXISTS (
      SELECT 1
      FROM subagent_spawns AS descendant
      WHERE descendant.parent_agent_id = sp.child_agent_id
        AND descendant.completed_at IS NULL
    )
    AND NOT EXISTS (
      SELECT 1
      FROM subagent_spawns AS sibling
      WHERE sibling.parent_agent_id = sp.parent_agent_id
        AND sibling.completed_at IS NULL
    )
    AND NOT EXISTS (
      SELECT 1
      FROM subagent_progress_tool_events AS tool_event
      WHERE tool_event.child_agent_id = sp.child_agent_id
        AND tool_event.delivered_at IS NULL
    )
    AND NOT EXISTS (
      SELECT 1
      FROM subagent_progress_message_events AS message_event
      WHERE message_event.child_agent_id = sp.child_agent_id
        AND message_event.delivered_at IS NULL
    )
    AND NOT EXISTS (
      SELECT 1
      FROM subagent_monitor_decisions AS decision
      WHERE decision.child_agent_id = sp.child_agent_id
        AND decision.created_at >= $3
    )
  ORDER BY sp.created_at ASC, sp.child_agent_id ASC
  LIMIT $4
  FOR UPDATE OF sp SKIP LOCKED
),

            c.child_status,
            c.parent_status,
            c.completed_at,
            c.prune_reason
