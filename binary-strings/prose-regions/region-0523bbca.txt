INSERT INTO meta_ai.media AS x (
  conversation_id, message_id, media_index, generation, media_key, kind, origin, mime_type,
  byte_size, file_name, caption, cache_state
)
SELECT $1, r.message_id, r.media_index, $2, r.media_key, r.kind, r.origin, r.mime_type,
       r.byte_size, r.file_name, r.caption, 'none'
FROM unnest($3::text[], $4::integer[], $5::text[], $6::text[], $7::text[], $8::text[],
            $9::bigint[], $10::text[], $11::text[])
  AS r(message_id, media_index, media_key, kind, origin, mime_type, byte_size, file_name,
       caption)
ON CONFLICT (conversation_id, message_id, media_index) DO UPDATE SET
  generation = excluded.generation,
  media_key = excluded.media_key,
  kind = excluded.kind,
  origin = excluded.origin,
  mime_type = excluded.mime_type,
  byte_size = excluded.byte_size,
  file_name = excluded.file_name,
  caption = excluded.caption,
  cache_state = CASE WHEN x.media_key = excluded.media_key OR x.cache_state = 'blocked'
                     THEN x.cache_state ELSE 'none' END,
  fetch_failures = CASE WHEN x.media_key = excluded.media_key THEN x.fetch_failures ELSE 0 END,
  last_fetch_error = CASE WHEN x.media_key = excluded.media_key THEN x.last_fetch_error END,
  retry_not_before = CASE WHEN x.media_key = excluded.media_key THEN x.retry_not_before END
WHERE (x.generation, x.media_key, x.kind, x.origin, x.mime_type, x.byte_size, x.file_name,
       x.caption)
  IS DISTINCT FROM
      (excluded.generation, excluded.media_key, excluded.kind, excluded.origin,
       excluded.mime_type, excluded.byte_size, excluded.file_name, excluded.caption)
