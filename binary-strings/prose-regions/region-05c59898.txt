WITH requested(citation_id) AS (
  SELECT DISTINCT citation_id
  FROM unnest($2::text[]) AS requested(citation_id)
), latest_event_per_citation AS (
  SELECT requested.citation_id, max(e.event_seq) AS event_seq
  FROM requested
  JOIN runtime.event_payload_fields AS entity_id
    ON entity_id.scalar_type = 'string'
   AND entity_id.text_value IS NOT NULL
   AND entity_id.field_path LIKE '/segments/%/segment/ads_entity/citation_id'
   AND hashtextextended(entity_id.text_value, 0) =
       hashtextextended(requested.citation_id, 0)
   AND entity_id.text_value = requested.citation_id
  JOIN runtime.events AS e
    ON e.event_seq = entity_id.event_seq
  WHERE e.parent_agent_id = $1
    AND e.event_name = 'message.assistant'
    AND e.visibility = 'public'
    AND e.role = 'assistant'
    AND e.payload_json IS NOT NULL
  GROUP BY requested.citation_id
), matching_events AS (
  SELECT DISTINCT event_seq
  FROM latest_event_per_citation
)
SELECT e.event_name, e.payload_json
FROM matching_events
JOIN runtime.events AS e USING (event_seq)
ORDER BY e.event_seq ASC

WITH requested(citation_id) AS (
  SELECT DISTINCT citation_id
  FROM unnest($2::text[]) AS requested(citation_id)
), latest_event_per_citation AS (
  SELECT requested.citation_id, max(e.event_seq) AS event_seq
  FROM requested
  JOIN runtime.event_payload_fields AS entity_id
    ON entity_id.scalar_type = 'string'
   AND entity_id.text_value IS NOT NULL
   AND (
     entity_id.field_path LIKE '/data/products/%/citation_id'
     OR entity_id.field_path LIKE '/segments/%/segment/product/citation_id'
   )
   AND hashtextextended(entity_id.text_value, 0) =
       hashtextextended(requested.citation_id, 0)
   AND entity_id.text_value = requested.citation_id
  JOIN runtime.events AS e
    ON e.event_seq = entity_id.event_seq
  WHERE e.parent_agent_id = $1
    AND e.event_name IN ('delta.presentation', 'message.assistant')
    AND e.visibility = 'public'
    AND e.role = 'assistant'
    AND e.payload_json IS NOT NULL
  GROUP BY requested.citation_id
), matching_events AS (
  SELECT DISTINCT event_seq
  FROM latest_event_per_citation
)
SELECT e.event_name, e.payload_json
FROM matching_events
JOIN runtime.events AS e USING (event_seq)
ORDER BY e.event_seq ASC

SELECT DISTINCT ON (e.message_id)
       e.message_id,
       m.body AS display_text,
       NULLIF(COALESCE(context_seq.text_value, context_seq.scalar_value), '')::bigint
         AS context_seq
FROM runtime.events AS e
LEFT JOIN runtime.messages AS m ON m.event_seq = e.event_seq
LEFT JOIN runtime.event_payload_fields AS context_seq
  ON context_seq.event_seq = e.event_seq
 AND context_seq.field_path = '/metadata/context_seq'
WHERE e.message_id = ANY($1::text[])
  AND e.event_name = 'message.assistant'
  AND e.visibility = 'public'
  AND e.role = 'assistant'
  AND e.transcript_surface = 'main_chat'
ORDER BY e.message_id, e.event_seq DESC
