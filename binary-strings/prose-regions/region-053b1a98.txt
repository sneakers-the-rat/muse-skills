-- The writer's row first; else the history owner's imported copy, which holds
-- the same provenance verbatim and outlives a finished call agent's rows.
WITH original AS (
  (SELECT p.provenance_json, 0 AS rank
   FROM runtime.events e
   JOIN agent.context_item_resume_projection p
     ON p.agent_id = COALESCE(NULLIF(e.agent_id, ''), e.parent_agent_id)
   JOIN runtime.event_payload_fields context_seq
     ON context_seq.event_seq = e.event_seq
    AND context_seq.field_path = '/metadata/context_seq'
    AND p.seq::text = COALESCE(context_seq.text_value, context_seq.scalar_value)
   WHERE e.message_id = $3 AND p.provenance_json IS NOT NULL
   LIMIT 1)
  UNION ALL
  (SELECT held.provenance_json, 1
   FROM agent.agents owner
   JOIN agent.context_item_resume_projection held
     ON held.agent_id = owner.presentation_root_session_id
   WHERE owner.id = $1
     AND held.provenance_json IS NOT NULL
     AND md5(agent.resume_projection_message_id_v1(held.provenance_json))::uuid
         BETWEEN md5($3)::uuid AND md5($3)::uuid
     AND agent.resume_projection_message_id_v1(held.provenance_json) = $3
   LIMIT 1)
), chosen AS (
  SELECT provenance_json FROM original ORDER BY rank LIMIT 1
)
UPDATE agent.context_item_resume_projection target
SET provenance_json = chosen.provenance_json, provenance_projected = true
FROM chosen

WITH candidates AS (
  SELECT sp.child_agent_id,
         sp.requester_source,
         COALESCE(child.status, sp.status) AS child_status,
         parent.status AS parent_status,
         sp.completed_at,
         CASE
           WHEN sp.requester_source = 'scheduler.cron'
             THEN 'scheduler_cron_stale_root'
           ELSE 'old_terminal_non_cron_root'
         END AS prune_reason
  FROM subagent_spawns AS sp
  JOIN agents AS parent
    ON parent.id = sp.parent_agent_id
  LEFT JOIN agents AS child
    ON child.id = sp.child_agent_id
  WHERE sp.completed_at IS NOT NULL
    AND sp.seen_at IS NULL
    AND parent.kind = 'root'
    AND (
      (
        sp.requester_source = 'scheduler.cron'
        AND sp.created_at <= $1
        AND sp.completed_at <= $1
        AND (
          parent.updated_at < sp.completed_at
          OR parent.status IN ('completed', 'errored', 'shutdown', 'not_found')
        )
      )
      OR (
        sp.requester_source IN ('runtime', 'runtime.browser_monitor')
        AND sp.created_at <= $2
        AND sp.completed_at <= $2
        AND COALESCE(child.status, sp.status) IN ('shutdown', 'interrupted', 'errored')
      )
    )
    AND NOT EXISTS (
      SELECT 1
      FROM subagent_spawns AS descendant
      WHERE descendant.parent_agent_id = sp.child_agent_id
        AND descendant.completed_at IS NULL
    )
    AND NOT EXISTS (
      SELECT 1
      FROM subagent_spawns AS sibling
      WHERE sibling.parent_agent_id = sp.parent_agent_id
        AND sibling.completed_at IS NULL
    )
    AND NOT EXISTS (
      SELECT 1
      FROM subagent_progress_tool_events AS tool_event
      WHERE tool_event.child_agent_id = sp.child_agent_id
        AND tool_event.delivered_at IS NULL
    )
    AND NOT EXISTS (
      SELECT 1
      FROM subagent_progress_message_events AS message_event
      WHERE message_event.child_agent_id = sp.child_agent_id
        AND message_event.delivered_at IS NULL
    )
    AND NOT EXISTS (
      SELECT 1
      FROM subagent_monitor_decisions AS decision
      WHERE decision.child_agent_id = sp.child_agent_id
        AND decision.created_at >= $3
    )
  ORDER BY sp.created_at ASC, sp.child_agent_id ASC
  LIMIT $4
  FOR UPDATE OF sp SKIP LOCKED
),
updated AS (
  UPDATE subagent_spawns AS sp
  SET seen_at = $5
  FROM candidates AS c
  WHERE sp.child_agent_id = c.child_agent_id
    AND sp.completed_at IS NOT NULL
    AND sp.seen_at IS NULL
  RETURNING c.requester_source,
            c.child_status,
            c.parent_status,
            c.completed_at,
            c.prune_reason
)

       child_status,

       completed_at,
       prune_reason
