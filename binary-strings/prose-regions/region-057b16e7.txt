
select input_grants_json
from runtime.browser_tasks
where task_id = $1
  and owner_kind is distinct from 'user'
  and admission_seq is not null
  and status = 'user_controlled'
  and owner_agent_id = $2
  and stream_owner_message_id = $3
  and state_revision = $4
  and outcome_status is null
for update
select 1 from runtime.browser_tasks where task_id = $1 for updatet.status = 'running' and t.last_progress_at <= now() - ($2::bigint * interval '1 second') and not exists (select 1 from runtime.browser_calls c where c.task_id = t.task_id and c.kind = 'action' and c.phase = 'dispatched')
select exists (
    select 1
    from runtime.browser_calls
    where operation_id = $1
      and kind = 'interaction'
      and result_json is not null
)

    terminal_reason = $6,
    terminal_interrupt_subtype = null,

    terminal_user_update_due_at = now(),
    outcome_status = $7,
    outcome_reason = null,
    outcome_at = now(),
    retention_end_reason = $8,
    requester_source = null,
    requester_transport = null,
    requester_model = null,
    requester_effective_model = null,
    request_mode_json = null,
    max_training_tier_json = null,
    initial_instruction = null,
    deadline_at = null,
    state_revision = state_revision + 1,
    updated_at = now()
where task_id = $1
  and admission_seq is not null
  and owner_agent_id = $2
  and stream_owner_message_id = $3
  and state_revision = $4
  and status = 'running'
  and outcome_status is null
  and outcome_reason is null
  and outcome_at is null
  and retention_end_reason is null
  and completed_at is null
