                     from feed.units u
                     join ideas.ideas i on i.idea_id = u.source_idea_id
                     where u.origin = 'idea'
                       and u.prompt_id = $1
                       and i.embedding is not null
                     order by u.edition_generated_at_ms desc, u.unit_id desc
                     limit $2select embedding::text as embedding_text
                     from feed.units
                     where embedding is not null
                       and prompt_id = $1
                       and origin <> 'idea'
                     order by created_at_ms desc, unit_id desc
                     limit $2select embedding::text as embedding_text
                     from feed.units
                     where embedding is not null
                       and prompt_id = $1
                       and origin <> 'idea'
