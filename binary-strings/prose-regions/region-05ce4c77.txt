SELECT e.event_seq,
       e.payload_json,
       message.body AS display_text
FROM runtime.events AS e
JOIN runtime.messages AS message ON message.event_seq = e.event_seq
WHERE e.parent_agent_id = $1
  AND e.reply_to_message_id = $2
  AND e.payload_json IS NOT NULL
  AND (
    (e.event_name = 'message.assistant' AND e.role = 'assistant' AND e.source = 'runtime')
    OR (e.event_name = 'message.internal' AND e.source = ANY($3::text[]))
  )
ORDER BY e.event_seq DESC
LIMIT 1
FOR UPDATE OF e
