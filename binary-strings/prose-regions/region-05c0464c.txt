-- These delivery and open aggregates are engagement telemetry only; briefing
-- eligibility is controlled by the goal's explicit `pushable` preference. A
-- briefing is "delivered in window" only if it carries a
-- `goal_briefing_push` impression within the lookback.
-- The `'goal_briefing_push'` surface literal must match
-- GOAL_BRIEFING_PUSH_SURFACE in the letter widget presentation writer and
-- `hatch_goals::GOAL_BRIEFING_PUSH_SURFACE` (edition delivery recording).
select cb.*,
       coalesce(agg.delivered_briefings_in_window, 0)::bigint as delivered_briefings_in_window,
       coalesce(agg.opened_delivered_in_window, 0)::bigint as opened_delivered_in_window,
       coalesce(agg.opened_briefings_in_window, 0)::bigint as opened_briefings_in_window
from candidate_briefings cb
left join lateral (
  select count(distinct prior.briefing_id) filter (where ev.goal_id is not null)::bigint
           as delivered_briefings_in_window,
         count(distinct prior.briefing_id) filter (
           where ev.goal_id is not null and prior.last_opened_at_ts >= $2
         )::bigint as opened_delivered_in_window,
         count(distinct prior.briefing_id) filter (
           where prior.last_opened_at_ts >= $2
         )::bigint as opened_briefings_in_window
  from goals.briefings prior
  left join goals.engagement_events ev
    on ev.goal_id = prior.goal_id
   and ev.surface = 'goal_briefing_push'
   and ev.metadata->>'briefing_id' = prior.briefing_id
   and ev.created_at >= $2
  where prior.goal_id = cb.briefing_goal_id
    and prior.briefing_id <> cb.briefing_briefing_id
    and not (prior.briefing_id = any($1::text[]))
) agg on true
order by cb.ordinality asc

                               null::text as last_delivered_at, last_opened_at
