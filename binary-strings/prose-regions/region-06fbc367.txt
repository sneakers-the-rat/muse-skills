WITH blockers AS (
  SELECT child_agent_id, bool_or(nonterminal) AS nonterminal,
         bool_or(completed_unseen) AS completed_unseen
  FROM (
    SELECT id AS child_agent_id, true AS nonterminal, false AS completed_unseen
    FROM agents
    WHERE kind = 'subagent'
      AND (status IS NULL OR status NOT IN ('completed', 'errored', 'shutdown', 'interrupted', 'not_found'))
      AND ($1::text IS NULL OR id > $1 OR id IS NULL)
    UNION ALL
    SELECT child_agent_id, false, true
    FROM subagent_spawns
    WHERE completed_at IS NOT NULL AND seen_at IS NULL
      AND (handoff_state_json -> 'undeliverable_result' ->> 'completed_at') IS DISTINCT FROM completed_at::text
      AND ($1::text IS NULL OR child_agent_id > $1)
  ) candidates
  GROUP BY child_agent_id
  ORDER BY child_agent_id NULLS LAST
  LIMIT $2
)
SELECT b.child_agent_id, b.nonterminal, b.completed_unseen,
       COALESCE(sp.parent_agent_id, a.parent_id) AS parent_agent_id,
       ancestry.root_agent_id,
       coordinator.coordinator_agent_id,
       a.status AS agent_status, sp.status AS spawn_status, p.status AS parent_status,
       a.depth AS agent_depth, sp.child_agent_id IS NOT NULL AS spawn_exists,
       p.id IS NOT NULL AS parent_exists, sp.completed_at, sp.seen_at,
       cp.mode AS checkpoint_mode, pcp.mode AS parent_checkpoint_mode,
       COALESCE(handoffs.rows, '[]'::jsonb) AS handoffs
FROM blockers b
LEFT JOIN agents a ON a.id = b.child_agent_id
LEFT JOIN subagent_spawns sp ON sp.child_agent_id = b.child_agent_id
LEFT JOIN agents p ON p.id = COALESCE(sp.parent_agent_id, a.parent_id)
LEFT JOIN runtime_restart_checkpoints cp ON cp.agent_id = b.child_agent_id
LEFT JOIN runtime_restart_checkpoints pcp ON pcp.agent_id = p.id

) ancestry ON true
LEFT JOIN LATERAL (
  -- Delegation depth is not structural depth: browser specialists inherit
  -- their planner's depth. Match the relay's spawn-parent edge to the root.
  SELECT CASE WHEN count(*) = 1 THEN min(ancestor.id) END AS coordinator_agent_id
  FROM agent_ancestors aa
  JOIN agents ancestor ON ancestor.id = aa.ancestor_id AND ancestor.kind = 'subagent'
  JOIN subagent_spawns ancestor_spawn ON ancestor_spawn.child_agent_id = ancestor.id
  WHERE aa.descendant_id = b.child_agent_id
    AND ancestor_spawn.parent_agent_id = ancestry.root_agent_id
    AND EXISTS (
      SELECT 1 FROM agent_ancestors root_path
      WHERE root_path.descendant_id = b.child_agent_id
        AND root_path.ancestor_id = ancestry.root_agent_id
        AND root_path.distance = aa.distance + 1
    )
) coordinator ON true
