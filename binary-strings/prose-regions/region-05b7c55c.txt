      SELECT 1
      FROM subagent_spawns AS descendant
      WHERE descendant.parent_agent_id = sp.child_agent_id
        AND descendant.completed_at IS NULL
    )
    AND NOT EXISTS (
      SELECT 1
      FROM subagent_spawns AS sibling

      FROM subagent_progress_tool_events AS tool_event
