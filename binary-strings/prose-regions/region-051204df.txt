event_transcript_rows AS (

SELECT
  e.event_seq AS seq,
  (extract(epoch from e.created_at) * 1000)::bigint AS occurred_at_ms,
  e.event_name,
  e.visibility::text AS visibility,
  coalesce(e.role::text, m.role::text, 'system') AS role,
  e.source,
  e.stream_lane,
  e.transcript_surface::text AS transcript_surface,
  e.request_id,
  e.root_request_id,
  e.parent_request_id,
  coalesce(e.message_id, m.message_id, '') AS message_id,
  e.reply_to_message_id,
  e.reply_target_message_id,
  e.agent_id,
  e.parent_agent_id,
  e.parent_message_id,
  case
    when derived_backlog.event_seq is not null
    then coalesce(e.payload_json, '{}'::text)
    else '{}'::text
  end AS payload_json,
  CASE
    WHEN e.event_name = 'delta.presentation' THEN NULLIF(m.body, '')
    ELSE m.body
  END AS display_text,
  case when e.display_text_ready then 1::bigint else 0::bigint end AS display_text_ready,
  null::bigint AS display_text_updated_at_ms,
  coalesce(e.idempotency_key, e.event_id::text) AS idempotency_key,
  channel_ctx.chat_context_json,
  coalesce(
    nullif(channel_ctx.chat_context_json::jsonb ->> 'origin_provider', ''),
    nullif(channel_ctx.chat_context_json::jsonb ->> 'provider', ''),
    external_transport.provider
  ) AS origin_provider,
  e.chat_kind,
  mr.reaction_emoji,
  coalesce(sd.search_text, m.body) AS search_text

FROM matched_chat_search_events AS matched
JOIN events e
  ON e.event_seq = matched.seq

LEFT JOIN messages m ON m.event_seq = e.event_seq
LEFT JOIN chat_event_derived_write_backlog derived_backlog ON derived_backlog.event_seq = e.event_seq
LEFT JOIN message_reactions mr ON mr.message_id = coalesce(e.message_id, m.message_id)
LEFT JOIN LATERAL (
  SELECT coalesce(
    (
      SELECT coalesce(ctx_field.text_value, ctx_field.scalar_value)
      FROM event_payload_fields ctx_field
      WHERE ctx_field.event_seq = e.event_seq
        AND ctx_field.field_path = '/__hatch/chat_context_json'
      LIMIT 1
    ),
    e.chat_context_json,
    nullif((e.payload_json::jsonb -> 'chat_context')::text, 'null')
  ) AS chat_context_json
) channel_ctx ON true
LEFT JOIN LATERAL (
  SELECT c.provider
  FROM chat.event_transports c
  WHERE c.event_seq = e.event_seq
    AND nullif(c.provider, '') IS NOT NULL
    AND c.provider <> 'main'
  ORDER BY c.provider ASC
  LIMIT 1
) external_transport ON true

LEFT JOIN search_documents sd ON sd.owner_table = 'runtime.events' AND sd.owner_key = e.event_seq::text

)
